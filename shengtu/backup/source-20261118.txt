    const POYIN_DB_URL = 'https://cdn.jsdelivr.net/gh/ButTaiwan/bpmfvs@master/ime/poyin_db.js';

      loaded: false,
      loadingPromise: null,
      phonicMap: new Map(), // char -> ['讀音0', '讀音1', ...]

    function stripIvs(str) {
      return String(str ?? '').replace(/[\u{E01E0}-\u{E01EF}]/gu, '');
    }

    // ① 第一次使用時用 fetch 載入兩個資料檔並快取
    async function ensureBpmfDictionariesLoaded() {
      if (bpmfDictState.loaded) return true;
      if (bpmfDictState.loadingPromise) return bpmfDictState.loadingPromise;

      bpmfDictState.loadingPromise = (async () => {
        try {
          const [phonicResp, poyinResp] = await Promise.all([
            fetch(PHONIC_TABLE_URL),
            fetch(POYIN_DB_URL)
          ]);

          if (!phonicResp.ok || !poyinResp.ok) {
            throw new Error('注音字典檔下載失敗');
          }

          const [phonicText, poyinText] = await Promise.all([
            phonicResp.text(),
            poyinResp.text()
          ]);

          // 解析 phonic_table_Z.txt：每行 tab 分隔「字、Unicode碼、來源、讀音0、讀音1...」
          const lines = phonicText.split(/\r?\n/);
          for (const line of lines) {
            if (!line || line.startsWith('#')) continue;
            const cols = line.split('\t');
            if (cols.length >= 4) {
              const ch = cols[0].trim();
              const readings = cols.slice(3).map(r => r.trim()).filter(Boolean);
              if (ch && readings.length > 0) {
                bpmfDictState.phonicMap.set(ch, readings);
              }
            }
          }

          // 解析 poyin_db.js：內容是 var data = {...}，用 new Function 取出 data 物件
          const extractDataFn = new Function(poyinText + '\nreturn typeof data !== "undefined" ? data : {};');
          bpmfDictState.poyinDb = extractDataFn() || {};

          bpmfDictState.loaded = true;
          return true;
        } catch (err) {
          console.warn('載入多音字資料表失敗，將使用預設讀音:', err);
          bpmfDictState.loadingPromise = null;
          return false;
        }
      })();

      return bpmfDictState.loadingPromise;
    }

    // ② 用詞語選音表 (poyin_db.js) 比對前後文，決定每個多音字的 n
    function matchPoyinDbReading(chars, index) {
      const ch = chars[index];
      const entry = bpmfDictState.poyinDb[ch];
      if (!entry || !entry.v) return 0;

      let bestN = 0;
      let bestPatternLen = 0;

      for (const [nKey, patternStr] of Object.entries(entry.v)) {
        const nVal = parseInt(nKey, 10);
        if (isNaN(nVal) || !patternStr) continue;
        const patterns = String(patternStr).split('/');

        for (const pat of patterns) {
          if (!pat) continue;
          const starOffset = Array.from(pat).indexOf('*');
          if (starOffset === -1) continue;

          const patChars = Array.from(pat);
          const startIdx = index - starOffset;
          if (startIdx < 0 || startIdx + patChars.length > chars.length) continue;

          let isMatch = true;
          for (let k = 0; k < patChars.length; k++) {
            if (k === starOffset) continue;
            if (chars[startIdx + k] !== patChars[k]) {
              isMatch = false;
              break;
            }
          }

          // 優先選取比對字數較長、較精確的詞型
          if (isMatch && patChars.length > bestPatternLen) {
            bestPatternLen = patChars.length;
            bestN = nVal;
          }
        }
      }

      return bestN;
    }

    function getReadingToneNumber(readingStr) {

      return 1;
    }

            selectedIndices[i] = findReadingIndexByTarget('角', 'ㄐㄧㄠˇ', selectedIndices[i]);
          }
        }

        // 17. 挑食、挑選、挑水、挑剔、挑毛病、挑擔子讀 ㄊㄧㄠ；挑戰、挑釁、挑撥、挑起讀 ㄊㄧㄠˇ
        if (ch === '挑') {
          if (['戰', '釁', '撥', '起', '動', '逗', '眉', '花', '燈'].includes(next)) {
            selectedIndices[i] = findReadingIndexByTarget('挑', 'ㄊㄧㄠˇ', selectedIndices[i]);
          } else {
            selectedIndices[i] = findReadingIndexByTarget('挑', 'ㄊㄧㄠ', selectedIndices[i]);
          }
        }
      }
    }

    // ③ 呼叫 Gemini 文字模型校正多音字（設定逾時保護，失敗或逾時就沿用步驟②的結果，絕不卡住）
    async function refinePolyphonesWithGemini(rawText, chars, polyphonePositions, selectedIndices) {
      if (polyphonePositions.length === 0) return;

      const candidateListDesc = polyphonePositions.map(pos => {
        const ch = chars[pos];
        const readings = bpmfDictState.phonicMap.get(ch) || [];
        const optionsStr = readings.map((r, idx) => `${idx}:${r}`).join(', ');
        return `位置 ${pos} 字「${ch}」候選讀音：[${optionsStr}]（預設 ${selectedIndices[pos]}）`;
      }).join('\n');

      const systemInstruction = `你是臺灣國小國語課本審定本的注音專家。請根據整段文字的前後文語境，為列出的多音字位置選出最正確的讀音編號（整數 n，從 0 起算）。
請嚴格遵守臺灣國小課本讀法規則：
1. 連接詞或介詞的「和」（如：我和你、小貓和小狗）讀 ㄏㄢˋ；和平、溫和讀 ㄏㄜˊ。
2. 爸爸、媽媽、哥哥、弟弟、妹妹、奶奶、伯伯、叔叔、爺爺的第二字讀輕聲（含 ˙ 的讀音）。
3. 意思、東西的第二字、名詞字尾的「子」（桌子、椅子、鞋子等）、石頭／磚頭／額頭／木頭的「頭」讀輕聲。
4. 馬虎的「虎」讀 ㄏㄨ。
5. 校長、班長、長高、長大讀 ㄓㄤˇ。
6. 因為、為了讀 ㄨㄟˋ；以為、成為讀 ㄨㄟˊ。
7. 跑得快、覺得的「得」與副詞後的「地」（慢慢地、開心地）讀 ㄉㄜ˙；得獎、得到讀 ㄉㄜˊ。
8. 互相讀 ㄒㄧㄤ。
9. 彈琴讀 ㄊㄢˊ。
10. 跌倒讀 ㄉㄠˇ。
11. 高興讀 ㄒㄧㄥˋ。
12. 音樂讀 ㄩㄝˋ。
13. 主角、角色讀 ㄐㄩㄝˊ。
14. 挑食、挑選讀 ㄊㄧㄠ。
請只回傳純 JSON 物件，格式為 {"位置索引": 讀音編號整數}，不要有任何其他說明文字或 Markdown。`;

      const userPrompt = `【完整文字】：\n${rawText}\n\n【需判斷的多音字位置與候選讀音】：\n${candidateListDesc}\n\n請回傳 JSON：`;

      const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;
      const controller = new AbortController();
      const timeoutId = setTimeout(() => controller.abort(), 5500);

      try {
        const resp = await fetch(apiUrl, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          signal: controller.signal,
          body: JSON.stringify({
            systemInstruction: { parts: [{ text: systemInstruction }] },

        if (parsedObj && typeof parsedObj === 'object') {

            }
          }
        }
      } catch (err) {

        // AI 失敗或逾時直接沿用步驟②與課本規則結果，不中斷流程
      }

    // ④ 程式規則處理「一」「不」的變調（以規則為準，覆蓋前面的結果）
    function applyYiBuToneSandhiRules(chars, selectedIndices) {
      const len = chars.length;
      const numberWords = '零〇一二兩三四五六七八九十百千萬億兆比點號年月日天週星期歲次樓層班級頁題號';
      const prefixKeepOriginalYi = '第同統唯初專單';
      const punctRegex = /^[\s\p{P}\p{S}]$/u;

      // 輔助：取得第 idx 字「已選定讀音」的聲調（若下一字是輕聲，依它的第 0 個讀音的聲調判斷）
      const getNextCharToneForSandhi = (nextIdx) => {
        if (nextIdx >= len) return null;
        const nextChar = chars[nextIdx];
        const nextReadings = bpmfDictState.phonicMap.get(nextChar);
        if (!nextReadings || nextReadings.length === 0) return null;

        const chosenN = selectedIndices[nextIdx] || 0;
        const chosenReading = nextReadings[chosenN] || nextReadings[0];
        let tone = getReadingToneNumber(chosenReading);

        // 下一字是輕聲 (5) 時，依它的第 0 個讀音的聲調判斷
        if (tone === 5) {
          tone = getReadingToneNumber(nextReadings[0]);
        }
        return tone;
      };

      for (let i = 0; i < len; i++) {
        const ch = chars[i];
        if (ch !== '一' && ch !== '不') continue;

        const prevChar = i > 0 ? chars[i - 1] : '';
        const nextChar = i + 1 < len ? chars[i + 1] : '';

        if (ch === '一') {
          // 檢查是否維持本調「ㄧ」：
          // 1. 後面是字串結尾或標點符號
          // 2. 前一字是「第」「同」「統」「唯」
          // 3. 前後是數字用字（二三四五六七八九十百千萬比點）或當題號（如「一、」「(一)」）
          const isEndOrPunct = !nextChar || punctRegex.test(nextChar);
          const isPrevSpecial = prevChar && prefixKeepOriginalYi.includes(prevChar);
          const isAroundNumber = (prevChar && '零〇一二兩三四五六七八九十百千萬億期週拜'.includes(prevChar)) || (nextChar && '零〇一二三四五六七八九十比月號'.includes(nextChar)) || (nextChar === '日' && prevChar === '月');
          const isListHeader = (i === 0 || prevChar === '\n' || prevChar === '（' || prevChar === '(') && (nextChar === '、' || nextChar === '.' || nextChar === '）' || nextChar === ')');

          if (isEndOrPunct || isPrevSpecial || isAroundNumber || isListHeader) {
            selectedIndices[i] = findReadingIndexByTarget('一', 'ㄧ', 0);
          } else {
            // 看下一個字已選定讀音的聲調：後面接四聲讀 ㄧˊ，後面接一、二、三聲讀 ㄧˋ（疊字中間如「讀一讀」「看一看」也照此規則）
            const nextTone = getNextCharToneForSandhi(i + 1);
            if (nextTone === 4) {
              selectedIndices[i] = findReadingIndexByTarget('一', 'ㄧˊ', 0);
            } else if (nextTone === 1 || nextTone === 2 || nextTone === 3) {
              selectedIndices[i] = findReadingIndexByTarget('一', 'ㄧˋ', 0);
            } else {
              selectedIndices[i] = findReadingIndexByTarget('一', 'ㄧ', 0);
            }
          }
        } else if (ch === '不') {
          // 「不」後面接四聲讀 ㄅㄨˊ，其他讀 ㄅㄨˋ
          const isEndOrPunct = !nextChar || punctRegex.test(nextChar);
          if (isEndOrPunct) {
            selectedIndices[i] = findReadingIndexByTarget('不', 'ㄅㄨˋ', 0);
          } else {
            const nextTone = getNextCharToneForSandhi(i + 1);
            if (nextTone === 4) {
              selectedIndices[i] = findReadingIndexByTarget('不', 'ㄅㄨˊ', 0);
            } else {
              selectedIndices[i] = findReadingIndexByTarget('不', 'ㄅㄨˋ', 0);
            }
          }
        }
      }
    }

    // 核心非同步函式 toBpmf(text)：回傳加好 IVS 的字串並快取結果
    async function toBpmf(text, options = { useAI: true }) {
      const cleanText = stripIvs(text);
      if (!cleanText || !/[\u4e00-\u9fff]/.test(cleanText)) {
        return cleanText;
      }

      const cacheKey = `${options.useAI ? 'AI' : 'FAST'}:${cleanText}`;
      if (bpmfResultCache.has(cacheKey)) {
        return bpmfResultCache.get(cacheKey);
      }
      // 若已有 AI 校正過的快取，直接共用
      if (bpmfResultCache.has(`AI:${cleanText}`)) {
        return bpmfResultCache.get(`AI:${cleanText}`);
      }

      const dictReady = await ensureBpmfDictionariesLoaded();
      if (!dictReady) {
        return cleanText;
      }

      const chars = Array.from(cleanText);
      const selectedIndices = new Array(chars.length).fill(0);
      const polyphonePositions = [];

      // 步驟②：先用詞語選音表比對前後文，決定每個多音字的 n
      for (let i = 0; i < chars.length; i++) {
        const ch = chars[i];
        const readings = bpmfDictState.phonicMap.get(ch);
        if (readings && readings.length > 1) {
          selectedIndices[i] = matchPoyinDbReading(chars, i);
          // 「一」與「不」由步驟④專屬變調規則處理，其餘多音字列入候選
          if (ch !== '一' && ch !== '不') {
            polyphonePositions.push(i);
          }
        }
      }

      // 套用國小課本核心讀音規則
      applyElementaryTextbookHeuristics(chars, selectedIndices);

      // 步驟③：呼叫 Gemini 文字模型校正多音字（失敗或逾時就沿用步驟②的結果）
      if (options.useAI && polyphonePositions.length > 0) {
        await refinePolyphonesWithGemini(cleanText, chars, polyphonePositions, selectedIndices);
        // 再次確保課本強制規則不被違反
        applyElementaryTextbookHeuristics(chars, selectedIndices);
      }

      // 步驟④：最後用程式規則處理「一」「不」的變調（以規則為準，覆蓋前面的結果）
      applyYiBuToneSandhiRules(chars, selectedIndices);

      // 組裝 IVS 字串：第 n 個讀音（n 從 0 算）在字後面加 String.fromCodePoint(0xE01E0 + n)，n === 0 時不加
      let resultStr = '';
      for (let i = 0; i < chars.length; i++) {
        const ch = chars[i];
        const n = selectedIndices[i] || 0;
        const readings = bpmfDictState.phonicMap.get(ch);
        if (readings && n > 0 && n < readings.length) {
          resultStr += ch + String.fromCodePoint(0xE01E0 + n);
        } else {
          resultStr += ch;
        }
      }

      bpmfResultCache.set(cacheKey, resultStr);
      return resultStr;
    }

    // 將指定 DOM 容器內的文字節點轉換為 IVS 注音（略過頂端工具列與輸入框）
    async function applyBpmfToDomContainer(rootEl, useAI = false) {
      if (!rootEl || !appState.ui.globalBpmf) return;

      const walker = document.createTreeWalker(
        rootEl,
        NodeFilter.SHOW_TEXT,
        {
          acceptNode(node) {
            const parent = node.parentElement;
            if (!parent) return NodeFilter.FILTER_REJECT;
            const tag = parent.tagName;
            if (tag === 'SCRIPT' || tag === 'STYLE' || tag === 'TEXTAREA' || tag === 'INPUT') {
              return NodeFilter.FILTER_REJECT;
            }
            if (parent.closest('#top-toolbar')) {
              return NodeFilter.FILTER_REJECT;
            }
            if (!node.nodeValue || !/[\u4e00-\u9fff]/.test(node.nodeValue)) {
              return NodeFilter.FILTER_REJECT;
            }
            return NodeFilter.FILTER_ACCEPT;
          }
        }
      );

      const textNodes = [];
      let current;
      while ((current = walker.nextNode())) {
        textNodes.push(current);
      }

      for (const node of textNodes) {
        const rawVal = stripIvs(node.nodeValue);
        domOriginalTextMap.set(node, rawVal);
        const converted = await toBpmf(rawVal, { useAI });
        if (appState.ui.globalBpmf && node.nodeValue !== converted) {
          node.nodeValue = converted;
        }
      }
    }

    // 關閉全頁注音時，還原 DOM 內所有文字節點為無 IVS 原文
    function restoreOriginalDomText(rootEl) {
      if (!rootEl) return;
      const walker = document.createTreeWalker(rootEl, NodeFilter.SHOW_TEXT);
      let current;
      while ((current = walker.nextNode())) {
        if (domOriginalTextMap.has(current)) {
          current.nodeValue = domOriginalTextMap.get(current);
        } else if (current.nodeValue && /[\u{E01E0}-\u{E01EF}]/u.test(current.nodeValue)) {
          current.nodeValue = stripIvs(current.nodeValue);
        }
      }
    }

    function parseTextIntoItems(silent = false) {
      const raw = stripIvs(document.getElementById('raw-text-input').value.trim());
      document.getElementById('raw-text-input').value = raw;
      appState.rawText = raw;

      if (!raw) {
        if (!silent) {
          showStatusBanner('請先在文字框輸入內容。', 'error');
        }
        return false;
      }

      // ① 依文字分組佇列保留舊項目物件（同一個 id、補充指示、畫面規劃、畫面重點），重複的詞按出現順序一對一配對
      const oldItemsPools = new Map();
      appState.items.forEach(it => {
        const key = stripIvs(it.text);
        if (!oldItemsPools.has(key)) {
          oldItemsPools.set(key, []);
        }
        oldItemsPools.get(key).push(it);
      });

      let parsedStrings = [];

      if (appState.mode === 'char') {
        const tokens = raw.split(/[\r\n、，,;\s]+/).map(s => s.trim()).filter(Boolean);
        tokens.forEach(tok => {
          const chars = Array.from(tok);
          chars.forEach(c => {
            if (c.trim()) parsedStrings.push(c.trim());
          });
        });
      } else if (appState.mode === 'word') {
        parsedStrings = raw.split(/[\r\n、，,;\s]+/).map(s => s.trim()).filter(Boolean);
      } else if (appState.mode === 'sentence') {
        parsedStrings = raw.split(/\r?\n/).map(s => s.trim()).filter(Boolean);
      } else if (appState.mode === 'story') {
        const lines = raw.split(/\r?\n/).map(s => s.trim()).filter(Boolean);
        if (lines.length === 1) {
          appState.storyTitle = '短文教材';
          parsedStrings = [lines[0]];
        } else {
          appState.storyTitle = lines[0];
          parsedStrings = lines.slice(1);
        }
        document.getElementById('story-title-input').value = appState.storyTitle;
        document.getElementById('story-title-box').classList.remove('hidden');
      }

      const nowTs = Date.now();
      appState.items = parsedStrings.map((txt, idx) => {
        const cleanTxt = stripIvs(txt);
        const pool = oldItemsPools.get(cleanTxt);
        if (pool && pool.length > 0) {
          // 沿用舊項目物件（同一個 id，保留補充指示、畫面規劃、畫面重點），只更新排序 index
          const matchedOldItem = pool.shift();
          matchedOldItem.index = idx + 1;
          matchedOldItem.text = cleanTxt;
          matchedOldItem.extraPrompt = stripIvs(matchedOldItem.extraPrompt || '');
          matchedOldItem.focusNoteZh = stripIvs(matchedOldItem.focusNoteZh || matchedOldItem.scenePlan?.noteZh || '');
          matchedOldItem.focusEditedByTeacher = Boolean(matchedOldItem.focusEditedByTeacher);
          return matchedOldItem;
        }
        // 只有新增的文字才建立新項目
        return {
          id: `item_${nowTs}_${idx}_${Math.random().toString(36).slice(2, 6)}`,
          index: idx + 1,
          text: cleanTxt,
          extraPrompt: '',
          scenePlan: null,
          focusNoteZh: '',
          focusEditedByTeacher: false
        };
      });

      renderParsedItemsList();
      syncItemsToCards();
      // 背景預先執行多音字與變調轉換並快取
      Promise.all(appState.items.map(it => toBpmf(it.text, { useAI: true }))).then(() => {
        if (appState.ui.globalBpmf) {
          refreshPageBpmfDisplay(false);
        }
      });
      return true;
    }

    function updateItemText(itemId, newText) {
      const cleanVal = stripIvs(newText);
      const item = appState.items.find(i => i.id === itemId);
      if (item) {
        if (item.text !== cleanVal) {
          item.scenePlan = null; // 文字改變時重新規劃畫面
          if (!item.focusEditedByTeacher) {
            item.focusNoteZh = '';
          }
        }
        item.text = cleanVal;
        appState.cards.filter(c => c.itemId === itemId).forEach(c => {
          c.text = cleanVal;
          c.scenePlan = item.scenePlan;
          c.focusNoteZh = item.focusNoteZh || '';
          c.focusEditedByTeacher = Boolean(item.focusEditedByTeacher);
        });
        renderCardsGrid();
      }
    }

    function updateItemFocusNote(itemId, newFocusNote) {
      const cleanVal = stripIvs(newFocusNote);

          c.focusNoteZh = cleanVal;

    /********************************************************************
     * 5. ④⑤⑥ 檔案匯入、資料夾匯入、Gemini 截圖文字辨識、語音輸入
     ********************************************************************/
    async function handleTextFilesUpload(event) {
      const files = Array.from(event.target.files || []);
      if (files.length === 0) return;

      const validExts = ['.txt', '.csv', '.md'];
      const validFiles = files.filter(f => {
        const lower = f.name.toLowerCase();
        return validExts.some(ext => lower.endsWith(ext));
      });

      if (validFiles.length === 0) {
        showStatusBanner('所選檔案中沒有 .txt、.csv 或 .md 文字檔。', 'error');
        event.target.value = '';
        return;
      }

      let combinedTexts = [];
      for (const file of validFiles) {
        try {
          const text = await file.text();
          if (text.trim()) combinedTexts.push(text.trim());
        } catch (err) {
          console.error('讀取文字檔錯誤:', err);
        }
      }

      const textarea = document.getElementById('raw-text-input');
      const currentVal = textarea.value.trim();
      textarea.value = currentVal
        ? currentVal + '\n' + combinedTexts.join('\n')
        : combinedTexts.join('\n');

      event.target.value = '';
      parseTextIntoItems(true);
    }

    function handleOcrFileInput(event) {
      const file = event.target.files?.[0];

    }

    // 支援整頁 Ctrl+V 直接貼上截圖辨識文字
    window.addEventListener('paste', (event) => {
      const clipboardItems = event.clipboardData?.items;
      if (!clipboardItems) return;
      for (const item of clipboardItems) {
        if (item.type && item.type.startsWith('image/')) {

          if (file) {

          }
        }
      }
    });

    async function recognizeTextFromImageFile(file) {
      const btn = document.getElementById('btn-ocr-upload');
      const txtSpan = document.getElementById('txt-ocr-upload');
      btn.disabled = true;
      txtSpan.textContent = '辨識中...';
      showStatusBanner('正在辨識圖片中的文字，請稍候...', 'wait');

      try {
        const dataUrl = await readFileAsDataURL(file);
        const base64Data = dataUrl.split(',')[1];
        const mimeType = file.type || 'image/png';

        const promptText = '請辨識這張圖片中的所有繁體中文文字（國小教材、國字、詞語、句子或短文）。請直接輸出辨識出的純文字內容，保持原本的換行或詞語分隔，不要加入任何解釋、問候語或 Markdown 符號。';
        const payload = {
          contents: [{
            role: 'user',
            parts: [
              { text: promptText },
              { inlineData: { mimeType: mimeType, data: base64Data } }
            ]
          }]
        };

        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;
        const result = await fetchWithBackoff(apiUrl, payload, 2);
        const recognizedText = result?.candidates?.[0]?.content?.parts?.map(p => p.text || '').join('').trim();

        if (recognizedText) {
          const textarea = document.getElementById('raw-text-input');
          const existing = textarea.value.trim();
          textarea.value = existing ? (existing + '\n' + recognizedText) : recognizedText;
          parseTextIntoItems(true);
          showStatusBanner('圖片文字辨識完成！已填入文字框。', 'done', 3000);
        } else {
          showStatusBanner('未能從圖片中辨識出文字，請確認圖片是否清晰。', 'error');
        }
      } catch (err) {
        showStatusBanner('圖片文字辨識失敗：' + (err.message || '網路或額度異常'), 'error');
      } finally {
        btn.disabled = false;
        txtSpan.textContent = '截圖辨識文字';
      }
    }

    // ⑥ 語音輸入按鈕（Web Speech API，zh-TW）
    function toggleVoiceRecognition() {
      const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
      if (!SpeechRecognition) {
        showStatusBanner('您的瀏覽器不支援語音輸入，建議使用 Chrome 或 Edge 瀏覽器。', 'error');
        return;
      }

      const btn = document.getElementById('btn-voice-input');
      const txt = document.getElementById('txt-voice-input');
      const icon = document.getElementById('icon-voice-input');

      if (isListeningVoice && speechRecognizer) {
        speechRecognizer.stop();
        return;
      }

      speechRecognizer = new SpeechRecognition();
      speechRecognizer.lang = 'zh-TW';
      speechRecognizer.continuous = true;
      speechRecognizer.interimResults = false;

      speechRecognizer.addEventListener('start', () => {
        isListeningVoice = true;
        btn.classList.add('is-active');
        btn.setAttribute('aria-pressed', 'true');
        icon.textContent = '🔴';
        txt.textContent = '停止語音';
      });

      speechRecognizer.addEventListener('result', (event) => {
        let transcript = '';
        for (let i = event.resultIndex; i < event.results.length; i++) {
          if (event.results[i].isFinal) {
            transcript += event.results[i][0].transcript;
          }
        }
        if (transcript.trim()) {
          const textarea = document.getElementById('raw-text-input');
          const cur = textarea.value.trim();
          const sep = (appState.mode === 'char' || appState.mode === 'word') ? '、' : '\n';
          textarea.value = cur ? (cur + sep + transcript.trim()) : transcript.trim();
        }
      });

      speechRecognizer.addEventListener('error', (event) => {
        showStatusBanner('語音輸入發生錯誤：' + event.error, 'error');
      });

      speechRecognizer.addEventListener('end', () => {
        isListeningVoice = false;
        btn.classList.remove('is-active');
        btn.setAttribute('aria-pressed', 'false');
        icon.textContent = '🎤';
        txt.textContent = '語音輸入';
      });

      speechRecognizer.start();
    }

    /********************************************************************
     * 7. 步驟三：圖卡資料同步、AI 畫面規劃與 Gemini 圖像生成排隊引擎
     ********************************************************************/
    function syncItemsToCards() {
      const count = appState.settings.countPerItem || 1;
      const newCards = [];

      appState.items.forEach((item, idx) => {
        item.index = idx + 1;
        for (let copy = 1; copy <= count; copy++) {
          const existing = appState.cards.find(
            c => c.itemId === item.id && c.copyIndex === copy
          );
          if (existing) {
            existing.itemIndex = item.index;
            existing.text = item.text;
            existing.extraPrompt = item.extraPrompt;
            existing.scenePlan = item.scenePlan || existing.scenePlan || null;
            existing.focusNoteZh = item.focusNoteZh || (existing.scenePlan && existing.scenePlan.noteZh) || existing.focusNoteZh || '';
            existing.focusEditedByTeacher = Boolean(item.focusEditedByTeacher);
            existing.totalCopies = count;
            newCards.push(existing);
          } else {
            newCards.push({
              cardId: `card_${item.id}_${copy}`,
              itemId: item.id,
              itemIndex: item.index,
              text: item.text,
              extraPrompt: item.extraPrompt,
              scenePlan: item.scenePlan || null,
              focusNoteZh: item.focusNoteZh || (item.scenePlan && item.scenePlan.noteZh) || '',
              focusEditedByTeacher: Boolean(item.focusEditedByTeacher),
              copyIndex: copy,
              totalCopies: count,
              status: 'idle',
              errorMsg: '',
              imageDataUrl: null,
              checked: true
            });
          }
        }
      });

      appState.cards = newCards;
      renderCardsGrid();
      refreshIdleBatchProgressUI();
    }

    function refreshIdleBatchProgressUI() {
      if (batchController.isRunning) return;
      const total = appState.cards.length;
      const doneCount = appState.cards.filter(c => Boolean(c.imageDataUrl)).length;
      const statusIconEl = document.getElementById('batch-status-icon');
      const statusLabelEl = document.getElementById('batch-status-label');
      const numbersEl = document.getElementById('batch-progress-numbers');
      const barEl = document.getElementById('batch-progress-bar');

      if (!statusLabelEl || !numbersEl || !barEl) return;

      if (total === 0) {
        if (statusIconEl) statusIconEl.textContent = '⏸️';
        statusLabelEl.textContent = '尚未開始';
        numbersEl.textContent = '已完成 0 張 ／ 共 0 張';
        barEl.style.width = '0%';
      } else if (doneCount === 0) {
        if (statusIconEl) statusIconEl.textContent = '⏸️';
        statusLabelEl.textContent = '尚未開始';
        numbersEl.textContent = `已完成 0 張 ／ 共 ${total} 張`;

      } else {

        statusLabelEl.textContent = doneCount === total ? '全部圖卡已完成' : '部分圖卡已完成';

        barEl.style.width = Math.round((doneCount / total) * 100) + '%';
      }

      if (appState.ui.globalBpmf) {

        if (box) applyBpmfToDomContainer(box, false);
      }
    }

    // 取得同批其他詞語清單（最多 30 個，去除自身與重複）
    function getOtherBatchWordsForText(currentText) {
      const cleanCur = stripIvs(currentText || '').trim();
      const others = [];
      for (const it of appState.items) {
        const w = stripIvs(it.text || '').trim();
        if (w && w !== cleanCur && !others.includes(w)) {
          others.push(w);
          if (others.length >= 30) break;
        }
      }
      return others;
    }

    // 固定規則保底產生器（當 AI 規劃逾時或失敗時使用，確保絕不卡住且符合特教詞卡聚焦原則與同批區分）
    function createFallbackScenePlan(itemText, mode, allBatchTexts = []) {
      const txt = stripIvs(itemText || '').trim();
      const otherWordsList = getOtherBatchWordsForText(txt);
      const otherAvoidClause = otherWordsList.length > 0
        ? `; strictly avoid looking like or sharing similar subject/composition/setting with other items in the same batch: [${otherWordsList.join(', ')}]`
        : '';

      const bodyPartRegex = /^(手指|手掌|手背|手腕|手臂|手肘|指甲|拳頭|腳趾|腳掌|腳踝|小腿|大腿|膝蓋|眼睛|眉毛|睫毛|鼻子|鼻孔|嘴巴|嘴唇|牙齒|舌頭|耳朵|耳垂|頭髮|額頭|臉頰|下巴|脖子|肩膀|肚子|肚臍|背部|屁股|手|腳|頭|臉|眼|耳|鼻|口|牙|舌)$/;
      const adultCategoryRegex = /^(大人|成年人|老師|爸爸|媽媽|父母|爺爺|奶奶|阿公|阿嬤|外公|外婆|伯伯|叔叔|阿姨|姑姑|舅舅|警察|醫生|護理師|廚師|司機|農夫|工人|老闆|店員|長輩|老人)$/;
      const childCategoryRegex = /^(小孩|孩子|小朋友|嬰兒|小寶寶|弟弟|妹妹|哥哥|姐姐|姊姊|同學|學生|男童|女童|男孩|女孩|幼兒)$/;
      const placeRegex = /(在家|家裡|屋裡|房間|客廳|廚房|浴室|廁所|臥室|陽台|學校|教室|操場|圖書館|保健室|走廊|公園|遊樂場|超市|超商|便利商店|市場|醫院|診所|郵局|銀行|車站|公車站|火車站|捷運站|動物園|博物館|海邊|山上|森林|馬路|街道|餐廳|麵包店|書店|玩具店|家|校|店|園|館)$/;
      const arriveActionRegex = /(走到|跑向|跑到|來到|抵達|到達|進到|走進|跑進|回到|回家|到校|到家|上車|下車|過馬路)/;

      if (mode === 'char' || mode === 'word') {
        if (bodyPartRegex.test(txt)) {
          const isHandPart = /手|指|掌|拳|腕|臂|肘/.test(txt);
          return {
            focus: `Macro close-up view of the specific human body part "${txt}" centered in the frame`,
            framing: '特寫',
            needPerson: isHandPart ? 'hand-only' : 'partial',
            mustShow: `Only the isolated body part "${txt}" filling 65% of the center frame with clear anatomical detail`,
            mustAvoid: `Full human body, human face (unless the word itself is face), pointing gesture that looks like "you", extra people, toys, background clutter${otherAvoidClause}`,
            noteZh: `特寫「${txt}」部位本身，不畫全身或臉部以免混淆`
          };
        }

        if (adultCategoryRegex.test(txt)) {
          return {
            focus: `A single person representing "${txt}" standing alone clearly in the center`,
            framing: '半身',
            needPerson: 'full',
            mustShow: `Strictly ONE single "${txt}" person with clear age and role features, occupying 65% of the frame`,
            mustAvoid: `Any children, kids, babies, or other age groups next to them (to avoid looking like "parents" or "family"), extra props${otherAvoidClause}`,
            noteZh: `單獨畫一位「${txt}」，不加小孩或其他年齡人物`
          };
        }

        if (childCategoryRegex.test(txt)) {
          return {
            focus: `A single child representing "${txt}" alone in the center`,
            framing: '半身',
            needPerson: 'full',
            mustShow: `Strictly ONE single "${txt}" occupying 65% of the frame`,
            mustAvoid: `Any adults, parents, teachers, or extra people around, distracting toys${otherAvoidClause}`,
            noteZh: `單獨畫一位「${txt}」，不加大人或其他人物`
          };
        }

        if (placeRegex.test(txt)) {
          const isHome = /在家|家裡|屋裡|家/.test(txt);
          return {
            focus: isHome
              ? `An immediately recognizable cozy house with a clear roof, door, and windows representing "${txt}", with a tiny calm figure visible inside or by the doorway`
              : `An immediately recognizable architectural landmark or environment representing the location "${txt}" as the main subject`,
            framing: '場景',
            needPerson: 'partial',
            mustShow: isHome
              ? 'Clear house structure with roof, walls, window, and doorway occupying the main focus; person is small inside the house and not doing distracting activities'
              : `Clear iconic visual features of the location "${txt}" occupying 70% of the frame; any person present is small and passive`,
            mustAvoid: `Close-up of a child playing with toys, eating, or doing prominent actions that steal focus from the location concept${otherAvoidClause}`,
            noteZh: `突顯「${txt}」場所建築特徵本身，不讓人物活動或玩具搶走焦點`
          };
        }

        if (arriveActionRegex.test(txt)) {
          return {
            focus: `Clear action of arriving at a distinct destination for "${txt}"`,
            framing: '全身',
            needPerson: 'full',
            mustShow: `A clear destination landmark (such as a doorway, gate, or target spot) AND the character right at the arrival point reaching/stepping into it to clearly depict "${txt}"`,
            mustAvoid: `Aimless walking in an empty street without a destination (must clearly differentiate from plain walking "走路")${otherAvoidClause}`,
            noteZh: `畫出明確終點目標與「抵達終點」的動作，與單純「走路」區分`
          };
        }

        // 一般物品 vs 一般動作判斷
        const commonActionChars = /(洗|刷|穿|脫|收|拿|放|抱|背|提|推|拉|開|關|吃|喝|咬|吞|吹|吸|看|聽|說|讀|寫|畫|唱|跳|跑|走|坐|站|躺|睡|醒|笑|哭|拍|揮|舉|指|摸|擦|掃|拖|洗|切|煮|炒|烤|晾|摺|疊|丟|撿|找|買|賣|送|給|借|還|排隊|敬禮|鞠躬|跌倒|爬|游|飛|騎|搭)/;
        if (commonActionChars.test(txt)) {
          return {
            focus: `Clear and unmistakable physical action of "${txt}" as the central visual focus`,
            framing: '半身',
            needPerson: 'full',
            mustShow: `Distinct action posture and key object directly involved in "${txt}", occupying 65% of the center frame`,
            mustAvoid: `Irrelevant background furniture, extra toys, extra people${otherAvoidClause}`,
            noteZh: `聚焦「${txt}」的關鍵動作與互動物品，背景單純`
          };
        }

        // 預設具體物品／自然名詞（不需要畫人物）
        return {
          focus: `The single concrete object or concept "${txt}" isolated and centered`,
          framing: '特寫',
          needPerson: 'none',
          mustShow: `Only the subject "${txt}" clearly centered, occupying 65%+ of the frame with vivid colors on a plain light background`,
          mustAvoid: `Any human characters, children, hands, toys, or unrelated background objects${otherAvoidClause}`,
          noteZh: `單獨聚焦「${txt}」主體，不畫人物與無關背景物`
        };
      }

      // 句子與短文模式預設保底
      return {
        focus: `The core action and event described in "${txt}"`,
        framing: '全身',
        needPerson: 'full',
        mustShow: `Clear depiction of the exact action in "${txt}" as the unmistakable focal point`,
        mustAvoid: `Unrelated background clutter or distracting side characters${otherAvoidClause}`,
        noteZh: `聚焦句中核心動作：「${txt.slice(0, 18)}」`
      };
    }

    // 【一、畫面規劃步驟】用 gemini-3-flash-preview 一次替整批項目規劃畫面（回傳 JSON；老師改過的項目不可覆蓋）
    async function planScenesForItems(itemsToPlan, forceReplanUnedited = false) {
      if (!itemsToPlan || itemsToPlan.length === 0) return;

      const allBatchTexts = appState.items.map(it => stripIvs(it.text));
      // 先為所有待規劃項目填入保底規則，但若老師已手動修改過畫面重點（focusEditedByTeacher），絕對保留老師寫的內容
      itemsToPlan.forEach(item => {
        const fallback = createFallbackScenePlan(item.text, appState.mode, allBatchTexts);
        if (!item.scenePlan) {
          item.scenePlan = fallback;
        } else {
          // 確保 mustAvoid 包含同批其他詞語
          const otherWords = getOtherBatchWordsForText(item.text);
          if (otherWords.length > 0 && item.scenePlan.mustAvoid && !item.scenePlan.mustAvoid.includes('same batch')) {
            item.scenePlan.mustAvoid += `; strictly avoid looking like other items in the same batch: [${otherWords.join(', ')}]`;
          }
        }
        if (item.focusEditedByTeacher && item.focusNoteZh) {
          item.scenePlan.noteZh = item.focusNoteZh;
        } else if (!item.focusNoteZh) {
          item.focusNoteZh = item.scenePlan.noteZh;
        }
      });

      const modeLabelMap = {
        char: '國字模式（每個國字一張詞義圖卡）',
        word: '詞語模式（每個詞語一張詞義圖卡）',
        sentence: '句子模式（每個句子一張情境動作圖卡）',
        story: `短文模式（題目：「${appState.storyTitle}」，每段一張故事圖卡）`
      };

      const itemsListDesc = itemsToPlan.map((it, idx) => {
        const extra = it.extraPrompt ? `（老師補充指示：${it.extraPrompt}）` : '';
        const teacherFocus = (it.focusEditedByTeacher && it.focusNoteZh) ? `（老師已指定畫面重點：${it.focusNoteZh}）` : '';
        return `${idx + 1}. id: "${it.id}", text: "${stripIvs(it.text)}" ${teacherFocus} ${extra}`;
      }).join('\n');

      const systemPrompt = `你是臺灣國小特殊教育（自閉症、智能障礙、語言發展遲緩學童）教材視覺設計專家。
請替老師輸入的整批教材項目規劃「最容易一眼看懂詞義、絕不會誤會混淆」的畫面構圖，並只回傳 JSON 陣列。

【核心視覺規劃原則（務必嚴格遵守）】：
1. 孩子一眼就要看出詞義：詞義主體放畫面正中央、佔畫面 60% 以上、色彩最鮮明；背景單純淡色，只有一個重點，絕不加與詞義無關的人、玩具、雜物。
2. 決定是否需要人物（needPerson 只能填 "none" | "hand-only" | "partial" | "full"）：
   - 具體物品、動物、植物、食物、自然現象（如：蘋果、書包、水、太陽、鉛筆）：needPerson 必須是 "none"（絕不畫人！）。
   - 身體部位詞（如：手指、手掌、指甲）：needPerson 填 "hand-only" 或 "partial"，framing 填 "特寫"，只畫該部位特寫，不畫全身、不畫臉，避免「手指」被誤會成「你」或「女孩」。
   - 人物類別詞（如：大人、小孩、老師、阿公、警察）：needPerson 填 "full"，但只畫那一類人「單獨一人」，絕不可在「大人」旁邊加小孩（否則學生會誤認成「父母／家人」）。
   - 地點詞（如：在家、學校、公園、超市）：必須畫出能一眼認出的「地點建築或環境本身」（例如「在家」：畫出有屋頂和門窗的房子外觀或清楚剖面，人物小小地在房子裡或門口），needPerson 填 "none" 或 "partial"，framing 填 "場景"；絕不可畫成小孩特寫在玩玩具而看不出地點！
   - 動作詞（如：走路 vs 走到、拿 vs 放）：要有極明確的動作線索；像「走到、跑到、回到」這類有終點的詞，mustShow 一定要畫出明確的終點目標（如門口、標竿、目的地）以及剛抵達終點的姿勢，並在 mustAvoid 註明不可和單純「走路」一樣沒有終點。
3. 同批區分（極重要）：同一批項目的主體、構圖（framing：特寫／半身／全身／場景）、視角與場景必須明顯不同，且每個項目的 mustAvoid 必須明確寫入「不可像同一批的其他詞語」，防止兩張圖看起來相似。
4. 若某項目老師已指定畫面重點，請完全依照老師指定的畫面重點來規劃英文 focus、framing、needPerson 與 mustShow。
5. 句子與短文模式：以「句中核心動作」為畫面焦點。`;

      const userPrompt = `【教材模式】：${modeLabelMap[appState.mode] || appState.mode}
【同一批的全部詞語清單（共 ${allBatchTexts.length} 項，每張圖的主體、構圖、場景絕不可與同批其他詞語混淆）】：${allBatchTexts.slice(0, 30).join('、')}
【老師設定的主題／情境】：${appState.settings.theme || '乾淨明亮背景，主體突出清楚'}

請為以下項目逐一規劃畫面，回傳 JSON Array：
${itemsListDesc}`;

      const payload = {
        systemInstruction: { parts: [{ text: systemPrompt }] },
        contents: [{ role: 'user', parts: [{ text: userPrompt }] }],
        generationConfig: {
          responseMimeType: 'application/json',
          temperature: 0.2,
          responseSchema: {
            type: 'ARRAY',
            items: {
              type: 'OBJECT',
              properties: {
                id: { type: 'STRING' },
                focus: { type: 'STRING', description: 'Main visual subject in English' },
                framing: { type: 'STRING', description: '特寫, 半身, 全身, or 場景' },
                needPerson: { type: 'STRING', description: 'none, hand-only, partial, or full' },
                mustShow: { type: 'STRING', description: 'Essential visual cues to include, in English' },
                mustAvoid: { type: 'STRING', description: 'Elements and other batch words to strictly exclude to prevent confusion, in English' },
                noteZh: { type: 'STRING', description: 'One concise Traditional Chinese sentence explaining the visual focus for the teacher' }
              },
              required: ['id', 'focus', 'framing', 'needPerson', 'mustShow', 'mustAvoid', 'noteZh']
            }
          }
        }
      };

      const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;
      const controller = new AbortController();
      const timeoutId = setTimeout(() => controller.abort(), 9000);

      try {
        const resp = await fetch(apiUrl, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          signal: controller.signal,
          body: JSON.stringify(payload)
        });
        clearTimeout(timeoutId);

        if (!resp.ok) return;
        const result = await resp.json();
        const jsonText = result?.candidates?.[0]?.content?.parts?.map(p => p.text || '').join('').trim();
        if (!jsonText) return;

        const cleaned = jsonText.replace(/^```json\s*/i, '').replace(/```$/i, '').trim();
        const parsedArr = JSON.parse(cleaned);
        if (!Array.isArray(parsedArr)) return;

        const validNeedPerson = ['none', 'hand-only', 'partial', 'full'];
        const validFraming = ['特寫', '半身', '全身', '場景'];

        parsedArr.forEach(planObj => {
          if (!planObj || !planObj.id) return;
          const item = appState.items.find(it => it.id === planObj.id);
          if (!item) return;

          const fallback = item.scenePlan || createFallbackScenePlan(item.text, appState.mode, allBatchTexts);
          const otherWords = getOtherBatchWordsForText(item.text);
          const batchAvoidSuffix = otherWords.length > 0
            ? ` (Strictly do NOT resemble any of the other batch items: ${otherWords.join(', ')})`
            : '';
          const rawMustAvoid = planObj.mustAvoid || fallback.mustAvoid;
          const combinedMustAvoid = rawMustAvoid.includes('other batch items') ? rawMustAvoid : (rawMustAvoid + batchAvoidSuffix);

          // 老師改過的項目，重新規劃時不可覆蓋其畫面重點文字（focusNoteZh）
          const preservedNoteZh = (item.focusEditedByTeacher && item.focusNoteZh)
            ? item.focusNoteZh
            : stripIvs(planObj.noteZh || fallback.noteZh);

          item.scenePlan = {
            focus: planObj.focus || fallback.focus,
            framing: validFraming.includes(planObj.framing) ? planObj.framing : fallback.framing,
            needPerson: validNeedPerson.includes(planObj.needPerson) ? planObj.needPerson : fallback.needPerson,
            mustShow: planObj.mustShow || fallback.mustShow,
            mustAvoid: combinedMustAvoid,
            noteZh: preservedNoteZh
          };
          if (!item.focusEditedByTeacher) {
            item.focusNoteZh = preservedNoteZh;
          }
        });
      } catch (err) {
        clearTimeout(timeoutId);
        // 規劃失敗或逾時直接沿用已建立的固定規則 fallback，絕不卡住
      } finally {
        // 同步規劃結果到所有圖卡
        appState.items.forEach(item => {
          appState.cards.filter(c => c.itemId === item.id).forEach(c => {
            c.scenePlan = item.scenePlan;
            c.focusNoteZh = item.focusNoteZh || (item.scenePlan && item.scenePlan.noteZh) || '';
            c.focusEditedByTeacher = Boolean(item.focusEditedByTeacher);
          });
        });
        renderParsedItemsList();
        renderCardsGrid();
      }
    }

    // 【一、老師修改的畫面重點最高優先】、【二、圈出重點】、【三、同批區分】組裝生圖提示詞
    function buildImagePromptForCard(card) {
      onSettingsChange();
      const s = appState.settings;
      const styleObj = STYLES_LIST.find(st => st.id === s.styleId) || STYLES_LIST[0];
      const styleBase = s.styleId === 'custom'
        ? (s.styleCustom || 'Clean educational illustration for elementary school students')
        : styleObj.prompt;

      // 取得該項目的畫面規劃與同批其他詞語清單（最多 30 個）
      const allBatchTexts = appState.items.map(it => stripIvs(it.text));
      const parentItem = appState.items.find(it => it.id === card.itemId);
      const plan = (parentItem && parentItem.scenePlan) || card.scenePlan || createFallbackScenePlan(card.text, appState.mode, allBatchTexts);
      const teacherFocusNote = stripIvs(
        (parentItem && parentItem.focusNoteZh) || card.focusNoteZh || (plan && plan.noteZh) || ''
      ).trim();
      const isTeacherCustomFocus = Boolean((parentItem && parentItem.focusEditedByTeacher) || card.focusEditedByTeacher);
      const otherBatchWords = getOtherBatchWordsForText(card.text);

      // 確保 mustAvoid 內含同一批的其他詞語清單
      let mustAvoidText = plan.mustAvoid || 'Background clutter, unrelated people or toys';
      if (otherBatchWords.length > 0 && !mustAvoidText.includes(otherBatchWords[0])) {
        mustAvoidText += `. Strictly avoid resembling any of the other items in the same batch: [${otherBatchWords.join(', ')}]`;
      }

      const framingEnMap = {
        '特寫': 'Macro Close-up / Tight Shot focusing strictly on the target subject',
        '半身': 'Medium / Half-body Shot centered on the key action or subject',
        '全身': 'Full-body Shot showing complete posture and clear movement',
        '場景': 'Wide / Establishing Scene Shot clearly showing the building or location structure'
      };
      const framingDesc = framingEnMap[plan.framing] || plan.framing || 'Clear centered shot';

      // 檢查是否為人物類別詞（大人、老師、阿公、嬰兒等），若是人物類別詞則不可套用固定的 8 歲學童外觀，以免把「大人」畫成小孩或加上小孩
      const cleanCardText = stripIvs(card.text || '').trim();
      const isSpecificPersonCategory = /^(大人|成年人|老師|爸爸|媽媽|父母|爺爺|奶奶|阿公|阿嬤|外公|外婆|伯伯|叔叔|阿姨|姑姑|舅舅|警察|醫生|護理師|廚師|司機|農夫|工人|老闆|店員|長輩|老人|嬰兒|小寶寶)$/.test(cleanCardText);

      let parts = [];
      parts.push(`Create a high-clarity educational flashcard illustration for elementary special education students in Taiwan.`);
      parts.push(`Art Style: ${styleBase}.`);
      if (s.styleExtra) {
        parts.push(`Additional Style Details: ${s.styleExtra}.`);
      }

      // 【一、老師修改「畫面重點」時，以老師改的文字為最高優先，寫進指令並要求依照它構圖】
      if (teacherFocusNote) {
        if (isTeacherCustomFocus) {
          parts.push(`=== HIGHEST PRIORITY: TEACHER'S CUSTOM VISUAL FOCUS & COMPOSITION INSTRUCTION ===`);
          parts.push(`The teacher has explicitly specified the exact visual focus and composition for this card: "${teacherFocusNote}".`);
          parts.push(`MANDATORY OVERRIDE: You MUST compose and draw the image strictly following "${teacherFocusNote}". If any general rule below conflicts with this instruction, "${teacherFocusNote}" takes absolute highest priority!`);
        } else {
          parts.push(`Key Visual Focus Summary (Traditional Chinese): "${teacherFocusNote}". Compose the scene strictly around this visual focus.`);
        }
      }

      // 【一、畫面規劃結果注入】
      parts.push(`=== VISUAL SCENE PLAN ===`);
      parts.push(`Target Concept: "${cleanCardText}"`);
      parts.push(`Primary Visual Focus (MUST be the unmistakable center of attention): ${plan.focus}.`);
      parts.push(`Camera Framing: ${framingDesc} (${plan.framing}).`);
      parts.push(`MUST SHOW (Essential Visual Cues): ${plan.mustShow}.`);
      parts.push(`MUST AVOID (Strictly Exclude to Prevent Confusion): ${mustAvoidText}.`);

      // 【三、同批區分：加入「同一批的其他詞語」清單（最多 30 個），要求這張圖的主體、構圖、場景不可像其中任何一個】
      if (otherBatchWords.length > 0) {
        parts.push(`=== BATCH DIFFERENTIATION RULE (OTHER WORDS IN THIS BATCH, MAX 30) ===`);
        parts.push(`Other items in the same batch: [${otherBatchWords.join('、')}].`);
        parts.push(`CRITICAL: The main subject, camera composition, posture, and scene setting of THIS image ("${cleanCardText}") MUST NOT look like or be confused with ANY of the other items listed above.`);
      }

      // 【二、圈出重點：勾選時指令要求用粗紅色圓圈或粗紅色箭頭標出詞義主體，圈外的元素淡化】
      if (s.highlightFocus) {
        parts.push(`=== VISUAL HIGHLIGHT / RED CIRCLE OR ARROW EMPHASIS RULE (MANDATORY) ===`);
        parts.push(`1. Draw a bold, bright RED circle (粗紅色圓圈) or a bold, bright RED arrow (粗紅色箭頭) directly around or pointing clearly at the exact target subject representing "${cleanCardText}" (for example, if the word is "adult 大人", circle the adult; if it is a body part or object, circle or point directly at that exact part/object).`);
        parts.push(`2. De-emphasize / fade out elements outside the red highlight: Make any surrounding context or elements outside the red circle/arrow noticeably lighter, softer, lower-contrast, and desaturated so the student's eyes lock immediately onto the highlighted target.`);
      }

      // 【二、詞卡聚焦規則：國字與詞語模式一律加入指令；句子與短文模式加入句中動作焦點】
      if (appState.mode === 'char' || appState.mode === 'word') {
        parts.push(`=== SPECIAL EDUCATION VOCABULARY CARD FOCUSING RULES (MANDATORY) ===`);
        parts.push(`1. Instant Meaning Recognition: This is a vocabulary card for special education children who need to grasp the exact meaning of "${cleanCardText}" at first glance.`);
        parts.push(`2. Single Dominant Subject (60%+ of Frame): Place the main subject of "${cleanCardText}" right in the center, occupying at least 60% to 75% of the frame, rendered with the most vivid and clear colors.`);
        parts.push(`3. Plain Uncluttered Background: Keep only ONE single focus. Use a simple, soft, plain light-colored background. DO NOT add unrelated people, toys, furniture, or decorative objects.`);
        parts.push(`4. Body Part Rule: If "${cleanCardText}" is a body part (e.g., finger, hand, foot, ear, eye), use a tight close-up showing ONLY that specific body part. DO NOT draw a full person or a human face, so students will not misinterpret it as "person" or "you".`);
        parts.push(`5. Person Category Rule: If "${cleanCardText}" refers to a category of person (e.g., adult, child, teacher, grandpa), draw ONLY ONE single person of that exact age/role alone. NEVER include people of other ages next to them (e.g., for "adult", never draw a child beside them, otherwise it looks like "parents").`);
        parts.push(`6. Location / Place Rule: If "${cleanCardText}" is a place or location (e.g., at home "在家", school, park), the recognizable building/place structure itself MUST be the main subject (e.g., for "at home 在家": draw a clear house with a roof, windows, and doorway, with a tiny calm person inside or at the doorway). Never let a character's activity or toys steal focus from the place itself.`);
        parts.push(`7. Action & Destination Rule: If "${cleanCardText}" is an action word, show unmistakable movement cues. For arrival/destination actions like "walk to (走到)", you MUST clearly draw the destination landmark (e.g., doorway, finish line, target spot) and the moment of arriving there, making it visually distinct from aimless "walking (走路)".`);
      } else {
        parts.push(`=== SENTENCE & STORY ACTION FOCUS RULE (MANDATORY) ===`);
        if (appState.mode === 'story') {
          parts.push(`Story Title: "${stripIvs(appState.storyTitle)}". Paragraph #${card.itemIndex}: "${cleanCardText}".`);
        }
        parts.push(`The exact action and event described in the sentence ("${cleanCardText}") MUST be the unmistakable central focal point of the illustration, occupying the center of the frame with a clean, non-distracting background.`);
      }

      if (s.theme) {
        parts.push(`General Background / Context Tone: ${s.theme}.`);
      }

      // 【三、角色一致改規則：只在需要人物時套用角色外觀，不需要人物就不畫人，身體部位只畫部位；若老師修改的畫面重點明確要求畫人，則允許畫人】
      const teacherExplicitlyWantsPerson = isTeacherCustomFocus && /(人|男|女|孩|童|師|爸|媽|爺|奶|公|嬤|哥|姐|姊|弟|妹|生|他|她)/.test(teacherFocusNote);
      const effectiveNeedPerson = teacherExplicitlyWantsPerson ? 'full' : plan.needPerson;

      if (effectiveNeedPerson === 'none') {
        parts.push(`HUMAN PRESENCE RULE (needPerson = none): This concept does NOT require any person. DO NOT draw any human characters, faces, or bodies anywhere in the image.`);
      } else if (effectiveNeedPerson === 'hand-only') {
        parts.push(`HUMAN PRESENCE RULE (needPerson = hand-only): Draw ONLY the hand or fingers in close-up as required by the concept. DO NOT draw a human face, head, torso, or full body.`);
      } else if (effectiveNeedPerson === 'partial') {
        if (placeRegexCheck(cleanCardText)) {
          parts.push(`HUMAN PRESENCE RULE (needPerson = partial / tiny scale): The location/building is the primary subject. If a person is shown inside or at the entrance, keep the person very small and calm${(s.charConsistent && s.charDesc && !isSpecificPersonCategory) ? ` (matching this outfit/look at a small scale: ${s.charDesc})` : ''}, never overshadowing the building.`);
        } else {
          parts.push(`HUMAN PRESENCE RULE (needPerson = partial): Show ONLY the specific body part required by "${cleanCardText}" in close-up. DO NOT show the full character.`);
        }
      } else {
        // effectiveNeedPerson === 'full'
        if (isSpecificPersonCategory) {
          parts.push(`HUMAN PRESENCE RULE: Depict strictly ONE single person matching the exact age and role of "${cleanCardText}" alone.`);
        } else if (s.charConsistent && s.charDesc) {
          parts.push(`Consistent Character Appearance (Apply because this scene requires a person): ${s.charDesc}.`);
          if (s.charRefDataUrl) {
            parts.push(`Reference Image Instruction: Match the character's facial features, hairstyle, and outfit from the attached reference image.`);
          }
        }
      }

      if (card.extraPrompt) {
        parts.push(`Teacher's Specific Additional Instruction for this card: ${card.extraPrompt}.`);
      }

      if (s.embedText) {
        parts.push(`Include the Traditional Chinese text "${cleanCardText}" clearly in the image.`);
      } else {
        parts.push(`CRITICAL REQUIREMENT: DO NOT include any text, letters, words, Chinese characters, numbers, watermarks, speech bubbles, or labels anywhere in the image. Purely visual illustration only.`);
      }

      return parts.join('\n');
    }

    function placeRegexCheck(txt) {
      return /(在家|家裡|屋裡|房間|客廳|廚房|浴室|廁所|臥室|陽台|學校|教室|操場|圖書館|保健室|走廊|公園|遊樂場|超市|超商|便利商店|市場|醫院|診所|郵局|銀行|車站|公車站|火車站|捷運站|動物園|博物館|海邊|山上|森林|馬路|街道|餐廳|麵包店|書店|玩具店|家|校|店|園|館)$/.test(txt);
    }

    async function generateSingleCardImage(card) {
      const promptText = buildImagePromptForCard(card);
      const parts = [{ text: promptText }];

      const allBatchTexts = appState.items.map(it => stripIvs(it.text));
      const plan = card.scenePlan || createFallbackScenePlan(card.text, appState.mode, allBatchTexts);
      const cleanCardText = stripIvs(card.text || '').trim();
      const isSpecificPersonCategory = /^(大人|成年人|老師|爸爸|媽媽|父母|爺爺|奶奶|阿公|阿嬤|外公|外婆|伯伯|叔叔|阿姨|姑姑|舅舅|警察|醫生|護理師|廚師|司機|農夫|工人|老闆|店員|長輩|老人|嬰兒|小寶寶)$/.test(cleanCardText);

      // 只有在畫面確實需要完整角色（且非特定其他年齡人物類別詞）時，才傳入角色參考圖，避免不該畫人的詞卡被強加主角
      const shouldAttachCharRef = Boolean(
        appState.settings.charRefDataUrl &&
        appState.settings.charConsistent &&
        plan.needPerson === 'full' &&
        !isSpecificPersonCategory
      );

      if (shouldAttachCharRef) {
        const base64Ref = appState.settings.charRefDataUrl.split(',')[1];
        if (base64Ref) {
          parts.push({
            inlineData: {
              mimeType: appState.settings.charRefMimeType || 'image/png',
              data: base64Ref
            }
          });
        }
      }

      const payload = {
        contents: [{ role: 'user', parts: parts }],
        generationConfig: {
          responseModalities: ['IMAGE'],
          imageConfig: {
            aspectRatio: appState.settings.aspectRatio || '1:1'
          }
        }
      };

      const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3.1-flash-image:generateContent?key=${apiKey}`;

      // 失敗時自動重試 2 次（共最多嘗試 3 次，間隔逐次加長：1.5秒、3.5秒）
      const retryDelays = [1500, 3500];
      let lastError = null;

      for (let attempt = 0; attempt <= 2; attempt++) {
        if (batchController.stopRequested) {
          throw new Error('使用者已停止生成');
        }
        try {
          const response = await fetch(apiUrl, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
          });

          if (!response.ok) {
            let errBody = '';
            try {
              const errJson = await response.json();
              errBody = errJson?.error?.message || JSON.stringify(errJson);
            } catch (_) {
              errBody = await response.text();
            }

            // 檢查是否為額度上限 (429 / RESOURCE_EXHAUSTED / quota)
            if (
              response.status === 429 ||
              /quota|rate_limit|resource_exhausted|too many requests/i.test(errBody)
            ) {
              const quotaErr = new Error(`已達到免費生圖額度上限（狀態碼 HTTP ${response.status}），請稍候幾分鐘再試，目前已完成的圖片皆已完整保留。`);
              quotaErr.isQuotaError = true;
              throw quotaErr;
            }

            const detailMsg = errBody ? `：${errBody.slice(0, 120)}` : '';
            throw new Error(`生圖請求失敗（狀態碼 HTTP ${response.status}）${detailMsg}`);
          }

          const result = await response.json();
          const candidate = result?.candidates?.[0];
          const imgPart = candidate?.content?.parts?.find(p => p.inlineData && p.inlineData.data);

          if (!imgPart) {
            const finishReason = candidate?.finishReason ? `，原因代碼：${candidate.finishReason}` : '';
            const blockReason = result?.promptFeedback?.blockReason ? `，安全攔截：${result.promptFeedback.blockReason}` : '';
            throw new Error(`模型未回傳影像資料（狀態碼 HTTP ${response.status}${finishReason}${blockReason}）`);
          }

          const rawDataUrl = `data:${imgPart.inlineData.mimeType || 'image/png'};base64,${imgPart.inlineData.data}`;
          // 轉成標準白底 JPG DataURL 儲存，確保預覽與下載皆一致為白底 JPG
          const whiteBgJpgUrl = await convertDataUrlToWhiteBgJpg(rawDataUrl);
          return whiteBgJpgUrl;
        } catch (err) {
          if (err.name === 'TypeError' && !err.message.includes('狀態碼')) {
            lastError = new Error(`網路連線異常或請求遭阻擋（狀態碼：網路連線錯誤，${err.message}）`);
          } else {
            lastError = err;
          }
          if (lastError.isQuotaError || batchController.stopRequested) {
            throw lastError;
          }
          if (attempt < 2) {
            await sleep(retryDelays[attempt]);
          }
        }
      }

      throw lastError || new Error('生圖請求失敗（狀態碼：未知錯誤，已自動重試 2 次）');
    }

    // 開始整批生成（結果區的「全部重新生圖」）
    async function startFullBatchGeneration() {
      if (batchController.isRunning) return;
      const raw = document.getElementById('raw-text-input').value.trim();
      if (appState.items.length === 0 || appState.cards.length === 0) {
        if (raw) {
          parseTextIntoItems(true);
        }
      }
      if (appState.cards.length === 0) {
        showStatusBanner('無法開始生圖：請先在上方文字框輸入教材文字！', 'error');
        return;
      }

      await runQueueForCards(appState.cards);
    }

    // 通用排隊執行器
    async function runQueueForCards(targetCards) {
      batchController.isRunning = true;
      batchController.stopRequested = false;

      const mainStartBtn = document.getElementById('btn-main-start');
      const startBtn = document.getElementById('btn-start-batch');
      const stopBtn = document.getElementById('btn-stop-batch');
      if (mainStartBtn) mainStartBtn.disabled = true;
      startBtn.disabled = true;
      stopBtn.disabled = false;

      const total = targetCards.length;
      let completedCount = 0;
      let errorCount = 0;
      let lastErrorText = '';
      let quotaStopped = false;

      // ② 先將所有排隊目標卡片設為「等待中」並立即渲染（保留既有的 card.imageDataUrl 不清空，直到新圖生成成功才覆蓋）
      targetCards.forEach(card => {
        card.status = 'waiting';
        card.errorMsg = '';
      });
      renderCardsGrid();

      const resultsSec = document.getElementById('results-section');
      if (resultsSec) {
        resultsSec.scrollIntoView({ behavior: 'smooth', block: 'start' });
      }

      // 【一、畫面規劃步驟】找出尚未規劃畫面的項目，先用 gemini-3-flash-preview 一次替整批項目規劃畫面（已規劃過的重新生成時直接沿用）
      const itemIdsInQueue = [...new Set(targetCards.map(c => c.itemId))];
      const itemsNeedingPlan = appState.items.filter(it => itemIdsInQueue.includes(it.id) && !it.scenePlan);
      if (itemsNeedingPlan.length > 0 && !batchController.stopRequested) {
        updateBatchProgressUI(0, total, '🧠 正在規劃每張圖卡的視覺重點與構圖...');
        await planScenesForItems(appState.items.filter(it => !it.scenePlan));
      } else {
        // 確保每張卡片都有對齊項目的 scenePlan
        const allBatchTexts = appState.items.map(it => stripIvs(it.text));
        targetCards.forEach(c => {
          const parentItem = appState.items.find(it => it.id === c.itemId);
          if (parentItem && parentItem.scenePlan) {
            c.scenePlan = parentItem.scenePlan;
          } else if (!c.scenePlan) {
            c.scenePlan = createFallbackScenePlan(c.text, appState.mode, allBatchTexts);
          }
        });
      }

      updateBatchProgressUI(0, total, '⏳ 正在排隊生成圖卡...');

      for (let i = 0; i < total; i++) {
        if (batchController.stopRequested) {
          // 被停止時，尚未開始的卡片恢復待生成或原本狀態
          for (let j = i; j < total; j++) {
            if (targetCards[j].status === 'waiting') {
              targetCards[j].status = targetCards[j].imageDataUrl ? 'success' : 'idle';
            }
          }
          break;
        }

        const card = targetCards[i];
        card.status = 'generating';
        card.errorMsg = '';
        renderCardsGrid();
        updateBatchProgressUI(i + 1, total, `🎨 生成中：「${card.text}」`);

        try {
          const jpgDataUrl = await generateSingleCardImage(card);
          card.imageDataUrl = jpgDataUrl;
          card.status = 'success';
          card.errorMsg = '';
          completedCount++;
        } catch (err) {
          if (batchController.stopRequested) {
            card.status = card.imageDataUrl ? 'success' : 'idle';
            for (let j = i + 1; j < total; j++) {
              if (targetCards[j].status === 'waiting') {
                targetCards[j].status = targetCards[j].imageDataUrl ? 'success' : 'idle';
              }
            }
            break;
          }
          card.status = 'error';
          card.errorMsg = err.message || '生圖請求失敗（狀態碼：未知）';
          errorCount++;
          lastErrorText = `「${card.text}」生圖失敗：${card.errorMsg}`;

          // 立即在上方訊息區用中文顯示錯誤原因（含狀態碼）
          showStatusBanner(lastErrorText, 'error');

          if (err.isQuotaError) {
            quotaStopped = true;
            for (let j = i + 1; j < total; j++) {
              if (targetCards[j].status === 'waiting') {
                targetCards[j].status = targetCards[j].imageDataUrl ? 'success' : 'idle';
              }
            }
            renderCardsGrid();
            break;
          }
        }

        renderCardsGrid();
      }

      batchController.isRunning = false;
      if (mainStartBtn) mainStartBtn.disabled = false;
      startBtn.disabled = false;
      stopBtn.disabled = true;

      const totalCardsCount = appState.cards.length;
      const totalDoneCount = appState.cards.filter(c => Boolean(c.imageDataUrl)).length;

      if (quotaStopped) {
        updateBatchProgressUI(totalDoneCount, totalCardsCount, '⚠️ 已達生圖額度上限，已保留完成圖卡', true);
      } else if (batchController.stopRequested) {
        updateBatchProgressUI(totalDoneCount, totalCardsCount, '⏹️ 已停止，已完成圖卡均保留', true);
      } else if (errorCount > 0) {
        updateBatchProgressUI(totalDoneCount, totalCardsCount, `⚠️ 完成 ${completedCount} 張，失敗 ${errorCount} 張`, true);
        showStatusBanner(`生圖結束：成功 ${completedCount} 張，失敗 ${errorCount} 張。最後錯誤原因：${lastErrorText}`, 'error');
      } else {
        refreshIdleBatchProgressUI();
      }

      // 自動將最新成果存入本機 IndexedDB
      saveCurrentProjectToDB(true);
    }

    /********************************************************************
     * 8. 步驟四：圖卡網格渲染、勾選、白底 JPG 轉換與檔名規則、JSZip 打包
     ********************************************************************/
    function getFormattedFilenameForCard(card) {
      // 檔名規則：一律用原文（不含 IVS）
      // 國字／詞語／句子模式：檔名就是該項文字，同一項有多張時在後面加 1、2、3
      // 短文模式：檔名是題目加段落序號（題目1、題目2）；若同一段有多張則加 -1, -2
      // 去除不合法的檔名字元，超過 40 字就截斷，副檔名 .jpg
      let base = '';
      if (appState.mode === 'story') {
        const cleanTitle = sanitizeFilename(stripIvs(appState.storyTitle || '短文'));
        base = `${cleanTitle}${card.itemIndex}`;
        if (card.totalCopies > 1) {
          base += `_${card.copyIndex}`;
        }
      } else {
        base = sanitizeFilename(stripIvs(card.text || `圖卡${card.itemIndex}`));
        if (base.length > 40) {
          base = base.slice(0, 40);
        }
        if (card.totalCopies > 1) {
          base = `${base}${card.copyIndex}`;
        }
      }

      if (base.length > 40) {
        base = base.slice(0, 40);
      }
      if (!base) {
        base = `圖卡_${card.itemIndex}`;
      }
      return `${base}.jpg`;
    }

    function sanitizeFilename(str) {
      return stripIvs(str)
        .replace(/[\\/:*?"<>|\r\n\t]+/g, '')
        .trim();
    }

    // 用 canvas 將任何圖片轉為白底 JPG (image/jpeg, 品質 0.92)
    function convertDataUrlToWhiteBgJpg(srcDataUrl) {
      return new Promise((resolve, reject) => {
        const img = new Image();
        img.crossOrigin = 'anonymous';
        img.addEventListener('load', () => {
          const canvas = document.createElement('canvas');
          canvas.width = img.naturalWidth || img.width || 1024;
          canvas.height = img.naturalHeight || img.height || 1024;
          const ctx = canvas.getContext('2d');
          // 填滿純白底色
          ctx.fillStyle = '#FFFFFF';
          ctx.fillRect(0, 0, canvas.width, canvas.height);
          ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
          const jpgUrl = canvas.toDataURL('image/jpeg', 0.92);
          resolve(jpgUrl);
        });
        img.addEventListener('error', () => reject(new Error('圖片載入 Canvas 轉換失敗')));
        img.src = srcDataUrl;
      });
    }

    // 下載勾選的或全部圖卡（一律使用 JSZip 打包成白底 .jpg）
    async function downloadCardsAsZip(onlySelected) {
      const targetCards = appState.cards.filter(c =>
        c.imageDataUrl && (onlySelected ? c.checked : true)
      );

      if (targetCards.length === 0) {
        showStatusBanner(
          onlySelected ? '勾選的圖卡中還沒有已生成的圖片可下載。' : '目前還沒有已生成的圖片可下載。',
          'error'
        );
        return;
      }

      showStatusBanner(`正在將 ${targetCards.length} 張圖片轉換為白底 JPG 並打包成 ZIP...`, 'wait');

      try {
        const zip = new JSZip();
        const usedNames = new Map();

        for (const card of targetCards) {
          // 再次確保轉為白底 JPG (image/jpeg, 品質 0.92)
          const jpgDataUrl = await convertDataUrlToWhiteBgJpg(card.imageDataUrl);
          const base64Data = jpgDataUrl.split(',')[1];

          let filename = getFormattedFilenameForCard(card);
          // 防止不同項目同名覆蓋
          if (usedNames.has(filename)) {
            const count = usedNames.get(filename) + 1;
            usedNames.set(filename, count);
            const dotIndex = filename.lastIndexOf('.jpg');
            filename = filename.slice(0, dotIndex) + `_${count}.jpg`;
          } else {
            usedNames.set(filename, 1);
          }

          zip.file(filename, base64Data, { base64: true });
        }

        const blob = await zip.generateAsync({ type: 'blob' });
        const projTitle = sanitizeFilename(document.getElementById('project-name-input').value || '特教圖卡');
        const zipFilename = `${projTitle}_圖卡(${targetCards.length}張).zip`;

        triggerBlobDownload(blob, zipFilename);

        showStatusBanner(`已成功打包並下載「${zipFilename}」！`, 'done', 3000);
      } catch (err) {
        showStatusBanner('打包 ZIP 失敗：' + err.message, 'error');
      }
    }

    // 統一使用 Blob + <a download> 下載檔案，絕不使用 window.open
    function triggerBlobDownload(blob, filename) {
      const url = URL.createObjectURL(blob);
      const link = document.createElement('a');
      link.style.display = 'none';
      link.href = url;
      link.setAttribute('download', filename);
      document.body.appendChild(link);
      link.click();
      setTimeout(() => {
        if (link.parentNode) {
          document.body.removeChild(link);
        }
        URL.revokeObjectURL(url);
      }, 1500);
    }

    // 根據每頁張數與直橫式計算最適網格欄列數
    function getGridColsRows(perPage, orient) {
      const isPortrait = orient === 'portrait';
      switch (Number(perPage)) {
        case 1: return { cols: 1, rows: 1 };
        case 2: return isPortrait ? { cols: 1, rows: 2 } : { cols: 2, rows: 1 };
        case 4: return { cols: 2, rows: 2 };
        case 6: return isPortrait ? { cols: 2, rows: 3 } : { cols: 3, rows: 2 };
        case 8: return isPortrait ? { cols: 2, rows: 4 } : { cols: 4, rows: 2 };
        case 9: return { cols: 3, rows: 3 };
        case 12: return isPortrait ? { cols: 3, rows: 4 } : { cols: 4, rows: 3 };
        default: return { cols: 2, rows: 2 };
      }
    }

      const container = document.getElementById('print-pages-container');
      const selectedCards = appState.cards.filter(c => c.checked);

      // 若圖卡文字加注音開啟，先將每張圖卡文字透過 toBpmf 轉為含 IVS 的字串
      const displayLabelMap = new Map();

          const bpmfTxt = await toBpmf(cleanTxt, { useAI: true });
          displayLabelMap.set(card.cardId, bpmfTxt);
        } else {
          displayLabelMap.set(card.cardId, cleanTxt);

      }

      const pages = [];
      for (let i = 0; i < selectedCards.length; i += cfg.perPage) {
        pages.push(selectedCards.slice(i, i + cfg.perPage));
      }

      const fontClass = cfg.useBpmf ? 'print-bpmf' : 'print-normal-font';

      container.innerHTML = pages.map((pageCards, pageIndex) => {
        const cellsHtml = pageCards.map(card => {
          const imgHtml = card.imageDataUrl

            : `<div style="flex:1; display:flex; align-items:center; justify-content:center; color:#666; font-size:14pt;">(尚未生成圖片)</div>`;

          const displayLabel = displayLabelMap.get(card.cardId) ?? stripIvs(card.text);
          const labelHtml = cfg.cardStyle === 'both'
            ? `<div class="print-card-label" style="font-size: ${cfg.fontSizePt}pt;">${escapeHtml(displayLabel)}</div>`
            : '';

          return `
            <div class="print-card-cell">
              ${imgHtml}

            </div>
          `;
        }).join('');

        return `
          <div class="no-print text-center font-bold mb-2 text-lg">
            📄 第 ${pageIndex + 1} 頁 ／ 共 ${pages.length} 頁（A4 ${cfg.orientation === 'portrait' ? '直式' : '橫式'}，每頁 ${cfg.perPage} 張）
          </div>
          <div
            class="print-sheet ${cfg.orientation} ${fontClass}"
            style="grid-template-columns: repeat(${cols}, minmax(0, 1fr)); grid-template-rows: repeat(${rows}, minmax(0, 1fr));"
          >
            ${cellsHtml}
          </div>
        `;
      }).join('');
    }

    // 載入單張圖片供 Canvas 繪製
    function loadImageElement(src) {
      return new Promise((resolve, reject) => {
        const img = new Image();
        img.crossOrigin = 'anonymous';
        img.addEventListener('load', () => resolve(img));
        img.addEventListener('error', () => reject(new Error('載入圖卡影像失敗')));
        img.src = src;
      });
    }

    // 計算文字自動換行與自動縮字以適應圖卡文字框（一律使用一般粗細 400，不加 bold）
    function layoutTextForCanvasCell(ctx, text, fontFamily, initialFontPx, maxWidth, maxHeight, lineHeightRatio) {
      let fontSize = initialFontPx;
      const minFontSize = 24; // 300dpi 下最小保底字級

      while (fontSize >= minFontSize) {
        ctx.font = `400 ${fontSize}px "${fontFamily}", "PingFang TC", "Microsoft JhengHei", sans-serif`;
        const lines = wrapCanvasTextLines(ctx, text, maxWidth);
        const lineHeight = fontSize * lineHeightRatio;
        const totalHeight = lines.length * lineHeight;

        if (totalHeight <= maxHeight || fontSize === minFontSize) {
          return { fontSize, lines, lineHeight, totalHeight };
        }
        fontSize -= 6;
      }

      const fallbackLines = wrapCanvasTextLines(ctx, text, maxWidth);
      return {
        fontSize: minFontSize,
        lines: fallbackLines,
        lineHeight: minFontSize * lineHeightRatio,
        totalHeight: fallbackLines.length * minFontSize * lineHeightRatio
      };
    }

    function wrapCanvasTextLines(ctx, text, maxWidth) {
      const rawParagraphs = String(text || '').split(/\r?\n/);
      const lines = [];

      for (const para of rawParagraphs) {
        // 將基礎字與緊隨其後的 IVS 選擇器 (U+E01E0 ~ U+E01EF) 視為同一個字元單位，不可拆開換行
        const rawCodePoints = Array.from(para);
        const graphemeUnits = [];
        for (const cp of rawCodePoints) {
          const code = cp.codePointAt(0);
          if (code >= 0xE01E0 && code <= 0xE01EF && graphemeUnits.length > 0) {
            graphemeUnits[graphemeUnits.length - 1] += cp;
          } else {
            graphemeUnits.push(cp);
          }
        }

        if (graphemeUnits.length === 0) continue;
        let currentLine = '';

        for (const unit of graphemeUnits) {
          const testLine = currentLine + unit;
          const metrics = ctx.measureText(testLine);
          if (metrics.width > maxWidth && currentLine.length > 0) {
            lines.push(currentLine);
            currentLine = unit;
          } else {
            currentLine = testLine;
          }
        }
        if (currentLine) {
          lines.push(currentLine);
        }
      }
      return lines.length > 0 ? lines : [''];
    }

    // 下載 300dpi A4 圖卡 PDF
    async function downloadA4CardsPDF() {
      const selectedCards = appState.cards.filter(c => c.checked);
      if (selectedCards.length === 0) {
        showStatusBanner('請先勾選至少一張要製作成 A4 PDF 的圖卡！', 'error');
        return;
      }

      if (!window.jspdf || !window.jspdf.jsPDF) {
        showStatusBanner('PDF 產生器（jsPDF）尚未載入完成，請檢查網路連線後再試一次。', 'error');
        return;
      }

      const btn = document.getElementById('btn-download-pdf');
      const iconEl = document.getElementById('icon-download-pdf');
      const txtEl = document.getElementById('txt-download-pdf');

      btn.disabled = true;
      iconEl.textContent = '⏳';

      try {
        const cfg = appState.printConfig;
        const isPortrait = cfg.orientation === 'portrait';
        // 300dpi A4 尺寸：直式 2480×3508、橫式 3508×2480
        const pageW = isPortrait ? 2480 : 3508;
        const pageH = isPortrait ? 3508 : 2480;
        const pageWmm = isPortrait ? 210 : 297;
        const pageHmm = isPortrait ? 297 : 210;

        // ⑥ 注音開啟時使用 BpmfIansuiFull，繪製前用 document.fonts.load 等它載入完成
        const targetFontFamily = cfg.useBpmf ? 'BpmfIansuiFull' : 'LXGW WenKai TC';
        txtEl.textContent = '載入字型中...';
        showStatusBanner('正在載入字型並校正多音字與變調讀音，準備繪製高解析度 A4 圖卡 PDF...', 'wait');

        // 先將所有圖卡文字透過 toBpmf 轉換為含 IVS 的字串（若開啟圖卡注音）
        const cardTextMap = new Map();
        for (const card of selectedCards) {
          const cleanTxt = stripIvs(card.text || '');
          if (cfg.useBpmf && cleanTxt) {
            cardTextMap.set(card.cardId, await toBpmf(cleanTxt, { useAI: true }));
          } else {
            cardTextMap.set(card.cardId, cleanTxt);
          }
        }

        if (document.fonts && document.fonts.load) {
          try {
            const sampleChars = Array.from(cardTextMap.values()).join('');
            await document.fonts.load(`400 48px "${targetFontFamily}"`, sampleChars || '測試注音字型');
            await document.fonts.ready;
          } catch (fontErr) {
            console.warn('字型預載提示:', fontErr);
          }
        }

        const { cols, rows } = getGridColsRows(cfg.perPage, cfg.orientation);
        const pages = [];
        for (let i = 0; i < selectedCards.length; i += cfg.perPage) {
          pages.push(selectedCards.slice(i, i + cfg.perPage));
        }

        const { jsPDF } = window.jspdf;
        const pdf = new jsPDF({
          orientation: isPortrait ? 'p' : 'l',
          unit: 'mm',
          format: 'a4',
          compress: true
        });

        // 10mm 邊距與 4mm 格距換算為 300dpi 像素 (1mm ≈ 11.811px)
        const marginPx = Math.round(10 * (2480 / 210)); // 約 118px
        const gapPx = Math.round(4 * (2480 / 210));     // 約 47px
        const cellPadPx = Math.round(3 * (2480 / 210)); // 約 35px

        const usableW = pageW - marginPx * 2;
        const usableH = pageH - marginPx * 2;
        const cellW = Math.floor((usableW - gapPx * (cols - 1)) / cols);
        const cellH = Math.floor((usableH - gapPx * (rows - 1)) / rows);

        // 300dpi 下 pt 轉 px 係數 (1pt = 1/72 inch, 300dpi 下 1pt = 300/72 ≈ 4.1667px)
        const baseFontPx = Math.round(cfg.fontSizePt * (300 / 72));
        const lineHeightRatio = cfg.useBpmf ? 1.85 : 1.38;

        for (let pIdx = 0; pIdx < pages.length; pIdx++) {
          const progressMsg = `正在製作 PDF 第 ${pIdx + 1} 頁 ／ 共 ${pages.length} 頁...`;
          txtEl.textContent = `製作中 (${pIdx + 1}/${pages.length})`;
          showStatusBanner(progressMsg, 'wait');
          // 讓 UI 有機會更新畫面進度
          await sleep(30);

          const canvas = document.createElement('canvas');
          canvas.width = pageW;
          canvas.height = pageH;
          const ctx = canvas.getContext('2d');

          // 整頁純白底色
          ctx.fillStyle = '#FFFFFF';
          ctx.fillRect(0, 0, pageW, pageH);

          const pageCards = pages[pIdx];
          for (let cIdx = 0; cIdx < pageCards.length; cIdx++) {
            const card = pageCards[cIdx];
            const col = cIdx % cols;
            const row = Math.floor(cIdx / cols);

            const x = marginPx + col * (cellW + gapPx);
            const y = marginPx + row * (cellH + gapPx);

            // 繪製每格細裁切虛線邊框
            ctx.save();
            ctx.strokeStyle = '#64748B';
            ctx.lineWidth = 3;
            ctx.setLineDash([16, 12]);
            ctx.strokeRect(x, y, cellW, cellH);
            ctx.restore();

            const innerX = x + cellPadPx;
            const innerY = y + cellPadPx;
            const innerW = cellW - cellPadPx * 2;
            const innerH = cellH - cellPadPx * 2;

            let imgBoxH = innerH;
            let textLayout = null;
            const renderedCardText = cardTextMap.get(card.cardId) ?? stripIvs(card.text || '');

            // 若樣式為「圖＋下方文字」，先計算文字所需高度與換行／自動縮小
            if (cfg.cardStyle === 'both' && renderedCardText) {
              const maxTextAreaH = Math.floor(innerH * 0.42);
              const textMarginW = cfg.useBpmf ? Math.floor(innerW * 0.9) : Math.floor(innerW * 0.94);
              textLayout = layoutTextForCanvasCell(
                ctx,
                renderedCardText,
                targetFontFamily,
                baseFontPx,
                textMarginW,
                maxTextAreaH,
                lineHeightRatio
              );
              const reservedTextH = Math.min(maxTextAreaH, Math.ceil(textLayout.totalHeight) + 24);
              imgBoxH = Math.max(80, innerH - reservedTextH);
            }

            // 繪製圖片（等比例置中不變形）
            if (card.imageDataUrl) {
              try {
                const img = await loadImageElement(card.imageDataUrl);
                const imgW = img.naturalWidth || img.width || 1024;
                const imgH = img.naturalHeight || img.height || 1024;
                const scale = Math.min(innerW / imgW, imgBoxH / imgH);
                const drawW = Math.round(imgW * scale);
                const drawH = Math.round(imgH * scale);
                const drawX = innerX + Math.round((innerW - drawW) / 2);
                const drawY = innerY + Math.round((imgBoxH - drawH) / 2);
                ctx.drawImage(img, drawX, drawY, drawW, drawH);
              } catch (imgErr) {
                ctx.save();
                ctx.fillStyle = '#94A3B8';
                ctx.font = `400 40px "${targetFontFamily}", sans-serif`;
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                ctx.fillText('(圖片載入失敗)', innerX + innerW / 2, innerY + imgBoxH / 2);
                ctx.restore();
              }
            } else {
              ctx.save();
              ctx.fillStyle = '#94A3B8';
              ctx.font = `400 44px "${targetFontFamily}", sans-serif`;
              ctx.textAlign = 'center';
              ctx.textBaseline = 'middle';
              ctx.fillText('(尚未生成圖片)', innerX + innerW / 2, innerY + imgBoxH / 2);
              ctx.restore();
            }

            // 繪製下方置中文字（一律使用一般粗細 400，不加 bold）
            if (cfg.cardStyle === 'both' && textLayout) {
              ctx.save();
              ctx.fillStyle = '#000000';
              ctx.font = `400 ${textLayout.fontSize}px "${targetFontFamily}", "ㄅ注音芫荽", "Bpmf Iansui", "PingFang TC", sans-serif`;
              ctx.textAlign = 'center';
              ctx.textBaseline = 'middle';

              const textRegionTop = innerY + imgBoxH;
              const textRegionH = innerH - imgBoxH;
              const startY = textRegionTop + (textRegionH - textLayout.totalHeight) / 2 + textLayout.lineHeight / 2;
              const centerX = innerX + innerW / 2;

              textLayout.lines.forEach((lineStr, lIdx) => {
                const lineY = startY + lIdx * textLayout.lineHeight;
                ctx.fillText(lineStr, centerX, lineY);
              });
              ctx.restore();
            }
          }

          // 將該頁 300dpi Canvas 轉為高品質 JPEG 加入 A4 PDF
          const pageJpegDataUrl = canvas.toDataURL('image/jpeg', 0.92);
          if (pIdx > 0) {
            pdf.addPage('a4', isPortrait ? 'p' : 'l');
          }
          pdf.addImage(pageJpegDataUrl, 'JPEG', 0, 0, pageWmm, pageHmm);
        }

        const rawProjName = document.getElementById('project-name-input').value.trim() || appState.projectName || '我的特教教材圖卡專案';
        const cleanProjName = sanitizeFilename(rawProjName) || '我的特教教材圖卡專案';
        const pdfFilename = `${cleanProjName}-圖卡.pdf`;

        const pdfBlob = pdf.output('blob');
        triggerBlobDownload(pdfBlob, pdfFilename);

        showStatusBanner(`已下載「${pdfFilename}」（共 ${pages.length} 頁），打開檔案即可直接列印！`, 'done', 3000);
      } catch (err) {
        console.error('產生 PDF 錯誤:', err);
        showStatusBanner('製作 PDF 失敗：' + (err.message || '未知錯誤'), 'error');
      } finally {
        btn.disabled = false;
        iconEl.textContent = '🖨️';
        txtEl.textContent = '下載 PDF';
      }
    }

    // 將 Base64 DataURL 轉為 Uint8Array 供 docx ImageRun 使用
    function dataUrlToUint8Array(dataUrl) {
      const base64 = dataUrl.split(',')[1] || '';
      const binaryString = window.atob(base64);
      const len = binaryString.length;
      const bytes = new Uint8Array(len);
      for (let i = 0; i < len; i++) {
        bytes[i] = binaryString.charCodeAt(i);
      }
      return bytes;
    }

    // 下載 A4 圖卡 Word 檔（.docx，可再編輯）
    async function downloadA4CardsDocx() {
      const selectedCards = appState.cards.filter(c => c.checked);
      if (selectedCards.length === 0) {
        showStatusBanner('請先勾選至少一張要製作成 Word 檔的圖卡！', 'error');
        return;
      }

      if (!window.docx) {
        showStatusBanner('Word 檔案產生器（docx 函式庫）尚未載入完成，請檢查網路連線後再試一次。', 'error');
        return;

      const btn = document.getElementById('btn-download-docx');
      const iconEl = document.getElementById('icon-download-docx');
      const txtEl = document.getElementById('txt-download-docx');

      try {

          Document,
          Packer,
          Paragraph,

          TableRow,

        } = window.docx;

        const cfg = appState.printConfig;

        const { cols, rows } = getGridColsRows(cfg.perPage, cfg.orientation);

        // ① A4 尺寸與 1 公分邊界換算為 twips (1 mm ≈ 56.6929 twips)
        // A4 短邊 210mm = 11906 twips，長邊 297mm = 16838 twips；1 公分 (10mm) = 567 twips
        const pageShortTwips = 11906;
        const pageLongTwips = 16838;
        const marginTwips = 567; // 1 公分

        // 實際紙張在指定方向下的寬高 (twips)
        const actualPageWTwips = isPortrait ? pageShortTwips : pageLongTwips;
        const actualPageHTwips = isPortrait ? pageLongTwips : pageShortTwips;

        // 扣除上下左右各 1 公分邊界後的可用區域，並預留約 240 twips 給表格外段落避免溢頁
        const usableWTwips = actualPageWTwips - marginTwips * 2;
        const usableHTwips = actualPageHTwips - marginTwips * 2 - 240;

        const cellWTwips = Math.floor(usableWTwips / cols);
        const rowHTwips = Math.floor(usableHTwips / rows);

        // 換算為 Word 96 DPI 下的像素 (1 twip = 96 / 1440 = 1 / 15 px)
        const cellPadTwips = 100; // 儲存格內距約 1.8mm
        const innerCellWPx = Math.max(40, Math.floor((cellWTwips - cellPadTwips * 2) / 15));
        const innerCellHPx = Math.max(40, Math.floor((rowHTwips - cellPadTwips * 2) / 15));

        // 【二、Word 檔字型名稱】注音文字的字型名稱（ascii、hAnsi、eastAsia、cs 四個都要）改成「ㄅ注音芫荽 Regular」，非注音時維持霞鶩文楷
        const fontName = cfg.useBpmf ? 'ㄅ注音芫荽 Regular' : 'LXGW WenKai TC';
        const fontSpec = {
          ascii: fontName,
          hAnsi: fontName,
          eastAsia: fontName,
          cs: fontName
        };
        const fontSizeHalfPt = Math.round(cfg.fontSizePt * 2); // docx size 單位為 half-point (1pt = 2)

        // 若開啟圖卡注音，先將每張圖卡文字透過 toBpmf 轉為含 IVS 的字串
        txtEl.textContent = '校正讀音中...';
        showStatusBanner('正在校正多音字與變調讀音，準備產生 Word 檔...', 'wait');
        const docxCardTextMap = new Map();
        for (const card of selectedCards) {
          const cleanTxt = stripIvs(card.text || '');
          if (cfg.useBpmf && cleanTxt) {
            docxCardTextMap.set(card.cardId, await toBpmf(cleanTxt, { useAI: true }));
          } else {
            docxCardTextMap.set(card.cardId, cleanTxt);
          }
        }

        // 細裁切框線設定
        const thinCutBorder = {
          style: BorderStyle.DASHED || BorderStyle.SINGLE,
          size: 4,
          color: '64748B'
        };
        const cellBorders = {
          top: thinCutBorder,
          bottom: thinCutBorder,
          left: thinCutBorder,
          right: thinCutBorder
        };

        // 依每頁張數分頁
        const pages = [];
        for (let i = 0; i < selectedCards.length; i += cfg.perPage) {
          pages.push(selectedCards.slice(i, i + cfg.perPage));
        }

        const docChildren = [];

        for (let pIdx = 0; pIdx < pages.length; pIdx++) {
          const progressMsg = `正在製作 Word 第 ${pIdx + 1} 頁 ／ 共 ${pages.length} 頁...`;
          txtEl.textContent = `製作中 (${pIdx + 1}/${pages.length})`;
          showStatusBanner(progressMsg, 'wait');
          await sleep(25);

          const pageCards = pages[pIdx];
          const tableRows = [];

          for (let r = 0; r < rows; r++) {
            const rowCells = [];
            for (let c = 0; c < cols; c++) {
              const cardIdx = r * cols + c;
              const card = pageCards[cardIdx];
              const cellChildren = [];

              if (card) {
                const renderedCardText = docxCardTextMap.get(card.cardId) ?? stripIvs(card.text || '');
                const hasLabel = cfg.cardStyle === 'both' && Boolean(renderedCardText);
                // 計算文字預留高度 (96 DPI 下 1pt ≈ 1.333px)
                const fontPx96 = Math.round(cfg.fontSizePt * (96 / 72));
                const approxCharsPerLine = Math.max(1, Math.floor(innerCellWPx / fontPx96));
                const textLen = Array.from(stripIvs(card.text || '')).length;
                const estLines = Math.max(1, Math.ceil(textLen / approxCharsPerLine));
                const lineFactor = cfg.useBpmf ? 1.95 : 1.45;
                const reservedTextHPx = hasLabel
                  ? Math.min(Math.floor(innerCellHPx * 0.45), Math.ceil(estLines * fontPx96 * lineFactor) + 14)
                  : 6;

                const maxImgWPx = Math.max(24, innerCellWPx - 8);
                const maxImgHPx = Math.max(24, innerCellHPx - reservedTextHPx - 8);

                if (card.imageDataUrl) {
                  try {
                    // 確保圖片轉為白底 JPG
                    const jpgDataUrl = await convertDataUrlToWhiteBgJpg(card.imageDataUrl);
                    const imgEl = await loadImageElement(jpgDataUrl);
                    const natW = imgEl.naturalWidth || imgEl.width || 1024;
                    const natH = imgEl.naturalHeight || imgEl.height || 1024;

                    const scale = Math.min(maxImgWPx / natW, maxImgHPx / natH);
                    const targetW = Math.max(16, Math.round(natW * scale));
                    const targetH = Math.max(16, Math.round(natH * scale));
                    const imgBytes = dataUrlToUint8Array(jpgDataUrl);

                    // ⑤ 圖片段落設定 keepNext，與下方文字保持同頁
                    cellChildren.push(
                      new Paragraph({
                        alignment: AlignmentType.CENTER,
                        keepNext: hasLabel,
                        keepLines: true,
                        spacing: { before: 40, after: hasLabel ? 60 : 40 },
                        children: [
                          new ImageRun({
                            data: imgBytes,
                            transformation: {
                              width: targetW,
                              height: targetH
                            }
                          })
                        ]
                      })
                    );
                  } catch (imgErr) {
                    cellChildren.push(
                      new Paragraph({
                        alignment: AlignmentType.CENTER,
                        keepNext: hasLabel,
                        spacing: { before: 60, after: 60 },
                        children: [
                          new TextRun({
                            text: '(圖片載入失敗)',
                            font: fontSpec,
                            size: 24,
                            color: '888888'
                          })
                        ]
                      })
                    );
                  }
                } else {
                  cellChildren.push(
                    new Paragraph({
                      alignment: AlignmentType.CENTER,
                      keepNext: hasLabel,
                      spacing: { before: 80, after: 80 },
                      children: [
                        new TextRun({
                          text: '(尚未生成圖片)',
                          font: fontSpec,
                          size: 24,
                          color: '888888'
                        })
                      ]
                    })
                  );
                }

                // ③ 樣式為「圖＋下方文字」時，加入可編輯的置中真文字段落（不加粗體 bold）
                if (hasLabel) {
                  const textLines = String(renderedCardText).split(/\r?\n/);
                  const textRuns = [];
                  textLines.forEach((lineText, idx) => {
                    textRuns.push(
                      new TextRun({
                        text: lineText,
                        font: fontSpec,
                        size: fontSizeHalfPt,
                        bold: false,
                        color: '000000',
                        break: idx > 0 ? 1 : 0
                      })
                    );
                  });

                  cellChildren.push(
                    new Paragraph({
                      alignment: AlignmentType.CENTER,
                      keepLines: true,
                      spacing: { before: 20, after: 20 },
                      children: textRuns
                    })
                  );
                }
              } else {
                // 空白補位格
                cellChildren.push(new Paragraph({ text: '' }));
              }

              rowCells.push(
                new TableCell({
                  width: { size: cellWTwips, type: WidthType.DXA },
                  verticalAlign: VerticalAlign.CENTER,
                  borders: cellBorders,
                  margins: {
                    top: cellPadTwips,
                    bottom: cellPadTwips,
                    left: cellPadTwips,
                    right: cellPadTwips
                  },
                  children: cellChildren
                })
              );
            }

            // ⑤ 每一列設定不可跨頁分割 (cantSplit: true)，⑥ 列高固定 (HeightRule.EXACT)
            tableRows.push(
              new TableRow({
                cantSplit: true,
                height: {
                  value: rowHTwips,
                  rule: HeightRule.EXACT
                },
                children: rowCells
              })
            );
          }

          // ② 每一頁一個表格，表格之間插入分頁符號
          if (pIdx > 0) {
            docChildren.push(
              new Paragraph({
                pageBreakBefore: true,
                spacing: { before: 0, after: 0 },
                children: []
              })
            );
          }

          docChildren.push(
            new Table({
              width: { size: usableWTwips, type: WidthType.DXA },
              columnWidths: Array(cols).fill(cellWTwips),
              layout: TableLayoutType ? TableLayoutType.FIXED : undefined,
              alignment: AlignmentType.CENTER,
              rows: tableRows
            })
          );
        }

        const doc = new Document({
          sections: [
            {
              properties: {
                page: {
                  size: {
                    width: pageShortTwips,
                    height: pageLongTwips,
                    orientation: isPortrait ? PageOrientation.PORTRAIT : PageOrientation.LANDSCAPE

    /********************************************************************
     * 10. 步驟六：IndexedDB 專案儲存、載入、刪除與 JSON 匯出入
     ********************************************************************/
    const DB_NAME = 'SpecialEdBatchImageHelperDB';
    const DB_VERSION = 1;
    const STORE_NAME = 'projects';

    function openProjectDB() {
      return new Promise((resolve, reject) => {
        const req = indexedDB.open(DB_NAME, DB_VERSION);
        req.addEventListener('upgradeneeded', (e) => {
          const db = e.target.result;
          if (!db.objectStoreNames.contains(STORE_NAME)) {
            db.createObjectStore(STORE_NAME, { keyPath: 'projectId' });
          }
        });
        req.addEventListener('success', () => resolve(req.result));
        req.addEventListener('error', () => reject(req.error));
      });
    }

    async function saveCurrentProjectToDB(silent = false) {
      onSettingsChange();
      appState.projectName = document.getElementById('project-name-input').value.trim() || '未命名特教專案';
      appState.rawText = document.getElementById('raw-text-input').value;

      const record = {
        projectId: appState.projectId,
        projectName: appState.projectName,
        updatedAt: new Date().toISOString(),
        stateSnapshot: JSON.parse(JSON.stringify(appState))
      };

      try {
        const db = await openProjectDB();
        await new Promise((resolve, reject) => {
          const tx = db.transaction(STORE_NAME, 'readwrite');
          tx.objectStore(STORE_NAME).put(record);
          tx.addEventListener('complete', () => resolve());
          tx.addEventListener('error', () => reject(tx.error));
        });
        await refreshSavedProjectsList();
      } catch (err) {
        if (!silent) {
          showStatusBanner('儲存專案失敗：' + err.message, 'error');
        }
      }
    }

    async function getAllSavedProjectsFromDB() {
      const db = await openProjectDB();
      return new Promise((resolve, reject) => {
        const tx = db.transaction(STORE_NAME, 'readonly');
        const req = tx.objectStore(STORE_NAME).getAll();
        req.addEventListener('success', () => resolve(req.result || []));
        req.addEventListener('error', () => reject(req.error));
      });
    }

    async function refreshSavedProjectsList() {
      try {
        const list = await getAllSavedProjectsFromDB();
        list.sort((a, b) => (b.updatedAt || '').localeCompare(a.updatedAt || ''));
        document.getElementById('saved-project-count').textContent = list.length;

        const container = document.getElementById('saved-projects-list');
        if (list.length === 0) {
          container.innerHTML = `<div class="py-3 font-bold" style="color: var(--text-muted);">目前尚無已儲存的專案。</div>`;
          return;
        }

        container.innerHTML = list.map(proj => {
          const dateStr = proj.updatedAt ? new Date(proj.updatedAt).toLocaleString('zh-TW') : '';
          const cardCount = proj.stateSnapshot?.cards?.length || 0;
          const doneCount = (proj.stateSnapshot?.cards || []).filter(c => c.imageDataUrl).length;
          return `
            <div class="app-card mb-2 py-3 px-4 flex flex-wrap items-center justify-between gap-3">
              <div>
                <div class="font-bold text-xl content-font">${escapeHtml(proj.projectName)}</div>
                <div class="helper-text" style="color: var(--text-muted);">
                  更新時間：${escapeHtml(dateStr)} ｜ 共 ${cardCount} 張（已完成 ${doneCount} 張）
                </div>
              </div>
              <div class="flex flex-wrap gap-2">
                <button type="button" data-action="load-project" data-id="${proj.projectId}" class="accessible-btn accessible-btn-primary">
                  <span aria-hidden="true">📂</span>
                  <span>載入</span>
                </button>
                <button type="button" data-action="copy-project" data-id="${proj.projectId}" class="accessible-btn">
                  <span aria-hidden="true">📋</span>
                  <span>另存副本</span>
                </button>
                <button type="button" data-action="delete-project" data-id="${proj.projectId}" class="accessible-btn accessible-btn-danger">
                  <span aria-hidden="true">🗑️</span>
                  <span>刪除</span>
                </button>
              </div>
            </div>
          `;
        }).join('');
      } catch (err) {
        console.error('載入專案列表失敗:', err);
      }
    }

    function toggleSavedProjectsPanel() {
      const panel = document.getElementById('saved-projects-panel');
      panel.classList.toggle('hidden');
      if (!panel.classList.contains('hidden')) {
        refreshSavedProjectsList();
      }
    }

    async function loadProjectFromDB(projectId) {
      try {
        const list = await getAllSavedProjectsFromDB();
        const found = list.find(p => p.projectId === projectId);
        if (!found || !found.stateSnapshot) {
          showStatusBanner('找不到該專案資料。', 'error');
          return;
        }
        applyLoadedState(found.stateSnapshot);
      } catch (err) {
        showStatusBanner('載入專案失敗：' + err.message, 'error');
      }
    }

    async function createNewProjectCopy(projectId) {
      try {
        const list = await getAllSavedProjectsFromDB();
        const found = list.find(p => p.projectId === projectId);
        if (!found) return;
        const copyState = JSON.parse(JSON.stringify(found.stateSnapshot));
        copyState.projectId = 'proj_' + Date.now();
        copyState.projectName = found.projectName + ' (副本)';
        applyLoadedState(copyState);
        await saveCurrentProjectToDB(true);
      } catch (err) {
        showStatusBanner('建立副本失敗：' + err.message, 'error');
      }
    }

    async function deleteProjectFromDB(projectId) {
      try {
        const db = await openProjectDB();
        await new Promise((resolve, reject) => {
          const tx = db.transaction(STORE_NAME, 'readwrite');
          tx.objectStore(STORE_NAME).delete(projectId);
          tx.addEventListener('complete', () => resolve());
          tx.addEventListener('error', () => reject(tx.error));
        });
        await refreshSavedProjectsList();
      } catch (err) {
        showStatusBanner('刪除專案失敗：' + err.message, 'error');
      }
    }

    // 匯入專案 JSON
    async function importProjectJSON(event) {
      const file = event.target.files?.[0];
      if (!file) return;
      try {
        const text = await file.text();
        const loaded = JSON.parse(text);
        if (!loaded || !Array.isArray(loaded.items)) {
          throw new Error('JSON 檔案格式不符合本程式專案結構');
        }
        loaded.projectId = loaded.projectId || ('proj_' + Date.now());
        applyLoadedState(loaded);
        await saveCurrentProjectToDB(true);
      } catch (err) {
        showStatusBanner('匯入專案檔失敗：' + err.message, 'error');
      }
      event.target.value = '';
    }

    function applyLoadedState(snapshot) {
      appState = Object.assign({}, appState, snapshot);
      if (Array.isArray(appState.items)) {
        appState.items.forEach(it => {
          it.focusNoteZh = it.focusNoteZh || (it.scenePlan && it.scenePlan.noteZh) || '';

      document.getElementById('setting-theme').value = appState.settings.theme || '';

      document.getElementById('setting-char-desc').value = appState.settings.charDesc || '';
      document.getElementById('setting-style-custom').value = appState.settings.styleCustom || '';

    /********************************************************************
     * 11. 通用工具函式
     ********************************************************************/
    function escapeHtml(str) {
      return String(str ?? '')
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#39;');
    }

    function sleep(ms) {
      return new Promise(res => setTimeout(res, ms));
    }

    function readFileAsDataURL(file) {
      return new Promise((resolve, reject) => {
        const reader = new FileReader();
        reader.addEventListener('load', () => resolve(reader.result));
        reader.addEventListener('error', () => reject(reader.error));
        reader.readAsDataURL(file);
      });
    }

    async function fetchWithBackoff(url, payload, retries = 2) {
      let delay = 1200;
      for (let i = 0; i <= retries; i++) {
        try {
          const resp = await fetch(url, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
          });
          if (!resp.ok) {
            throw new Error(`HTTP ${resp.status}`);
          }
          return await resp.json();
        } catch (err) {
          if (i === retries) throw err;
          await sleep(delay);
          delay *= 2;
        }
      }
    }

    /********************************************************************
     * 12. 全域與動態元素事件委派綁定（完全符合嚴格 CSP，無任何行內事件屬性）
     ********************************************************************/
    function bindAllEvents() {
      // 1. 全頁按鈕與動作點擊事件委派
      document.addEventListener('click', (event) => {
        const actionBtn = event.target.closest('[data-action]');
        if (!actionBtn || actionBtn.disabled) return;

        const action = actionBtn.dataset.action;
        // 避免針對 input 的 data-action 在 click 時誤觸
        if (action === 'update-item-text' || action === 'update-item-focus' || action === 'update-item-extra' || action === 'toggle-card-select') {
          return;
        }

        switch (action) {
          case 'set-font-size':
            setRootFontSize(Number(actionBtn.dataset.size));
            break;
          case 'toggle-hc':
            toggleHighContrast();
            break;
          case 'toggle-dark':
            toggleDarkMode();
            break;
          case 'toggle-bpmf':
            toggleGlobalBpmf();
            break;
          case 'save-project':
            saveCurrentProjectToDB(false);
            break;
          case 'toggle-saved-projects':
            toggleSavedProjectsPanel();
            break;
          case 'export-project':
            exportProjectJSON();
            break;
          case 'trigger-file': {
            const targetId = actionBtn.dataset.target;
            const fileInput = targetId ? document.getElementById(targetId) : null;
            if (fileInput) fileInput.click();
            break;
          }
          case 'load-project':
            loadProjectFromDB(actionBtn.dataset.id);
            break;
          case 'copy-project':
            createNewProjectCopy(actionBtn.dataset.id);
            break;
          case 'delete-project':
            deleteProjectFromDB(actionBtn.dataset.id);
            break;
          case 'set-input-mode':
            setInputMode(actionBtn.dataset.mode);
            break;
          case 'toggle-voice':
            toggleVoiceRecognition();
            break;
          case 'insert-sample':
            insertSampleText();
            break;
          case 'clear-textarea':
            clearRawTextarea();
            break;
          case 'start-from-input':
            startFromInputDirectly();
            break;
          case 'preview-items':
            previewAndEditItems();
            break;
          case 'plan-scenes-now':
            triggerManualPlanScenes();
            break;
          case 'add-blank-item':
            addBlankItem();
            break;
          case 'delete-item':
            deleteItem(actionBtn.dataset.id);
            break;
          case 'clear-char-ref':
            clearCharRefImage();
            break;
          case 'select-style':
            selectStyle(actionBtn.dataset.id);
            break;
          case 'select-ratio':
            selectAspectRatio(actionBtn.dataset.ratio);
            break;
          case 'select-count':
            selectCountPerItem(Number(actionBtn.dataset.count));
            break;
          case 'start-full-batch':
            startFullBatchGeneration();
            break;
          case 'stop-batch':
            stopBatchGeneration();
            break;
          case 'toggle-select-all':
            toggleSelectAllCards();
            break;
          case 'regenerate-selected':
            regenerateSelectedCards();
            break;
          case 'regenerate-single':
            regenerateSingleCard(actionBtn.dataset.id);
            break;
          case 'download-zip':
            downloadCardsAsZip(actionBtn.dataset.onlySelected === 'true');
            break;
          case 'open-print-layout':
            openPrintLayoutSection();
            break;
          case 'close-print-layout':
            closePrintLayoutSection();
            break;
          case 'set-print-orient':
            setPrintOrientation(actionBtn.dataset.orient);
            break;
          case 'set-print-style':
            setPrintCardStyle(actionBtn.dataset.style);
            break;
          case 'toggle-print-bpmf':
            togglePrintBpmf();
            break;
          case 'set-print-perpage':
            setPrintPerPage(Number(actionBtn.dataset.num));
            break;
          case 'download-pdf':
            downloadA4CardsPDF();
            break;
          case 'download-docx':
            downloadA4CardsDocx();
            break;
          case 'scroll-to-top':
            scrollToTopSmooth();
            break;
          case 'hide-status-banner':
            hideStatusBanner();
            break;
        }
      });

          const target = event.target;
          if (!target || !target.dataset) return;
          if (target.dataset.action === 'update-item-text') {
            updateItemText(target.dataset.id, target.value);
          } else if (target.dataset.action === 'update-item-focus') {
            updateItemFocusNote(target.dataset.id, target.value);
          } else if (target.dataset.action === 'update-item-extra') {
            updateItemExtraPrompt(target.dataset.id, target.value);
